```typescript
import type { PdfPageText } from './types';
export interface PdfCell { text: string; x0: number; x1: number }
export interface PdfLine { page: number; y: number; cells: PdfCell[] }
export const LINE_TOLERANCE = 2;    // pt: baselines this close share a line
export const CELL_JOIN_GAP = 4;     // pt: runs this close on a line join into one cell
export const HEADER_LINE_GAP = 14;  // pt: max distance between the two lines of a split header
export function groupLines(pages: readonly PdfPageText[]): PdfLine[];
export function looksNumeric(text: string): boolean;
export type ColumnRole = 'date' | 'description' | 'amount' | 'debit' | 'credit' | 'balance' | 'other';
export const HEADER_VOCABULARY: Readonly<Record<Exclude<ColumnRole, 'other'>, readonly string[]>>;
export function matchHeaderCell(text: string): Exclude<ColumnRole, 'other'> | null;
export interface ColumnAnchor { role: ColumnRole; x0: number; x1: number }   // no header text is carried (D-14)
export interface HeaderMatch { line: number; span: 1 | 2; anchors: ColumnAnchor[]; availableLabel: boolean }
export function findHeaders(lines: readonly PdfLine[]): HeaderMatch[];
export type Placement = { kind: 'column'; role: ColumnRole } | { kind: 'between'; roles: ['debit', 'credit'] };
export function placeCell(cell: PdfCell, anchors: readonly ColumnAnchor[]): Placement;
```
HEADER_VOCABULARY (exact whole-cell matches after normalising: upper-case, whitespace collapsed, trailing ':' removed, a trailing currency qualifier '(£)', '($)', '(€)', '(GBP)', '(USD)', '(EUR)' or ' £' removed): date ['DATE', 'TRANSACTION DATE', 'TRANS DATE', 'POSTING DATE', 'POST DATE', 'POSTED', 'VALUE DATE']; description ['DESCRIPTION', 'DETAILS', 'TRANSACTION DETAILS', 'TRANSACTION', 'PARTICULARS', 'NARRATIVE', 'MERCHANT', 'PAYEE']; amount ['AMOUNT', 'VALUE']; debit ['DEBIT', 'DEBITS', 'PAID OUT', 'MONEY OUT', 'WITHDRAWALS', 'WITHDRAWAL', 'OUT']; credit ['CREDIT', 'CREDITS', 'PAID IN', 'MONEY IN', 'DEPOSITS', 'DEPOSIT', 'RECEIPTS', 'IN']; balance ['BALANCE', 'RUNNING BALANCE', 'AVAILABLE BALANCE', 'AVAILABLE CREDIT']. 'PAYMENTS' and 'CHARGES' are deliberately absent (their direction differs between bank and card statements).
groupLines: per page, items sorted by y descending then x ascending; a new line starts when |y − line.y| > LINE_TOLERANCE; cells joined when `next.x − (prev.x + prev.w) <= CELL_JOIN_GAP` (joined with '' when the gap < 0.5, else ' '); cell text whitespace-collapsed and trimmed.
looksNumeric: one anchored regex without nested quantifiers accepting an optional leading sign/paren/currency symbol, digits with ',', '.', ' ', U+00A0, U+202F or "'" separators, an optional closing paren, and an optional trailing ' OD' | ' CR' | ' DR' | 'CR' | 'DR' | '-' (case-insensitive); must contain at least one digit; dates like '20-30-40' and '02 Mar' must fail (require the last digit group to be preceded by '.' or ',' followed by exactly 2 digits, OR be a plain digit run of <= 12 digits).
findHeaders: a line is a header when >= 3 of its cells match, 'date' is among them, at least one of amount/debit/credit is among them, and no matched role repeats; unmatched cells on that line become 'other' anchors. When a line has >= 2 matches but fails, merge it with the next line on the same page if within HEADER_LINE_GAP: cells whose x ranges overlap by >= 50% of the narrower are joined 'TOP BOTTOM', others kept; test the merged set (span 2). availableLabel = a matched balance cell's normalised text starts with 'AVAILABLE'.
placeCell: numeric cells consider only amount/debit/credit/balance anchors, distance = min(|cell.x1 − a.x1|, |mid(cell) − mid(a)|); text cells consider date/description/other anchors, distance = |cell.x0 − a.x0|. For a numeric cell whose two nearest anchors are debit and credit and `(d2 − d1) < 0.25 * |mid(debit) − mid(credit)|` -> between; otherwise the nearest anchor's role. Append `export * from './layout';` to pdf/index.ts. RED then GREEN.

Phrase matching uses `includes` on the upper-cased joined line text for OPENING/CLOSING/'CREDIT LIMIT'/'AVAILABLE CREDIT' (capture the line's last numeric cell text) and `startsWith` on the first text cell for REMOVED_PHRASES; these checks run before row detection on every line, header or not. periodLine: the first line containing 'PERIOD' or whose text has two date-like tokens joined by ' to ' / ' - ' / ' – '. Hints are booleans from phrases ('SORT CODE', 'ACCOUNT NUMBER', 'IBAN', 'ROUTING' -> bank; 'CREDIT LIMIT', 'CARD ENDING', 'CARD NUMBER', 'MINIMUM PAYMENT' -> card); no digits from those lines are kept anywhere.
Header mode: lines before the first header only feed summary/period/hints; each HeaderMatch's anchors apply until the next header; data-line rules in order: (1) a placed date cell for which `looksLikeDate` is true starts a row; (2) otherwise a line with a numeric cell placed in amount/debit/credit/between starts a row flagged 'no-date'; (3) otherwise a line of only text cells continues the previous row (descriptionParts push, 'wrapped' flag once) — if there is no previous row it is ignored; (4) a line with only a balance value and no previous balance on the previous row attaches to it; else ignored. Numeric placements fill amountText/debitText/creditText/balanceText; 'between' sets amountColumn 'between' and flag 'between'; both debit and credit present -> 'both-columns'. roles = distinct anchor roles of the first header excluding 'other'.
