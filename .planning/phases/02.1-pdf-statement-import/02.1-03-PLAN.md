export const STATEMENT_BUCKET = 'statement-uploads';
export const PDF_MAX_PAGES = 25;                    // D-07
export const PDF_MAX_BYTES = 10 * 1024 * 1024;      // D-07, equals the bucket file_size_limit 10485760

export interface PdfTextItem { str: string; x: number; y: number; w: number }  // x = left edge, y = baseline (PDF units, origin bottom-left), w = run width
export interface PdfPageText { n: number; width: number; height: number; items: PdfTextItem[] }

export type PdfReadResponse =
  | { ok: true; pageCount: number; pages: PdfPageText[] }
  | { ok: false; error: PdfReadError; pageCount?: number };

`read.ts` (imports only './contract.ts' and './logMeta.ts'; header: D-10 delete in the same request, D-13 nothing written server-side, D-15 metadata-only log, D-21 password used once):
```typescript
export interface RawTextItem { str: string; x: number; y: number; w: number }
export interface PdfDoc { numPages: number; page(n: number): Promise<{ width: number; height: number; items: RawTextItem[] }>; destroy(): Promise<void> }
export type PdfOpenResult = { kind: 'ok'; doc: PdfDoc } | { kind: 'password-required' } | { kind: 'password-incorrect' } | { kind: 'invalid' };
export interface ReadDeps {
  userId(): Promise<string | null>;                 // from the caller's verified JWT
  download(path: string): Promise<Uint8Array | null>;  // service-role Storage download; null when absent
  remove(path: string): Promise<void>;              // service-role Storage remove
  openPdf(bytes: Uint8Array, password: string | null): Promise<PdfOpenResult>;
  now(): number;
  log(line: string): void;
}
export type ReadResult = { status: 200 | 400 | 401 | 500; body: PdfReadResponse };
export function normalizePage(n: number, raw: { width: number; height: number; items: RawTextItem[] }): PdfPageText;
export async function readStatement(deps: ReadDeps, body: unknown): Promise<ReadResult>;
```
Order: validate body (object; `path` string matching STATEMENT_PATH_RE; optional `password` string 1..PDF_PASSWORD_MAX) -> 400 → `uid = await deps.userId()` -> 401 when null → owner = first capture group of the path; owner !== uid -> 200 'not-found' with NO download/remove → `try { download; null -> not-found; length > PDF_MAX_BYTES -> too-large; openPdf(bytes, password ?? null); map non-ok kinds; numPages > PDF_MAX_PAGES -> too-many-pages (before any page() call); for n = 1..numPages: normalizePage(n, await doc.page(n)) accumulating item count, > PDF_MAX_ITEMS -> unreadable; zero items across all pages -> no-text-layer; else ok } catch (err) { errorClass = errorClassOf(err); a download/remove failure -> 500 'internal' (errorClass 'StorageError'); a parse failure (open/page) -> 200 'unreadable' } finally { try { await deps.remove(path); removed = true } catch { removed = false }; await doc?.destroy().catch(() => {}); deps.log(buildLogLine({ … })) }`. Domain outcomes are HTTP 200 with `ok: false` so the app's functions.invoke never has to parse an HTTP error. The password variable is never placed in any object other than the openPdf call. normalizePage: drop items whose `str.trim() === ''`; `str` trimmed and sliced to PDF_MAX_ITEM_CHARS; numbers rounded with `Math.round(v * 10) / 10`; non-finite numbers drop the item.

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| app -> read-pdf-statement | Untrusted JSON body (path, password) from an authenticated caller |
| Storage object -> PDF parser | Fully untrusted, possibly hostile PDF bytes |
| function -> logs / response | Anything logged or returned leaves the isolate |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-02.1-03-01 | Elevation of Privilege (IDOR) | path parameter | mitigate | Strict path regex; owner segment must equal the JWT user id; foreign paths return not-found with no download and no delete |
| T-02.1-03-02 | Information Disclosure (retention) | uploaded object | mitigate | remove() in finally on every owned outcome, including thrown errors (D-10); removal failure logged as a flag for the D-12 purge to catch |
| T-02.1-03-03 | Information Disclosure (logs) | log line | mitigate | Single allow-list builder; closed-list error class from err.name only; no-leak test with fixture strings, path and password (D-15) |
| T-02.1-03-04 | Information Disclosure (error echo) | response body | mitigate | Fixed error codes only; IN-B01 catch never forwards err.message |
| T-02.1-03-05 | Denial of Service | hostile PDF | mitigate | Byte cap before open; page cap before any page parse; item cap; refuse rather than truncate (D-07) |
| T-02.1-03-06 | Information Disclosure (password) | D-21 password | mitigate | Used only as the openPdf argument; never logged, stored or returned; test asserts |
| T-02.1-03-07 | Tampering (silent truncation) | page loop | mitigate | >25 pages is an explicit refusal, never a silent cut (grep gate) |
