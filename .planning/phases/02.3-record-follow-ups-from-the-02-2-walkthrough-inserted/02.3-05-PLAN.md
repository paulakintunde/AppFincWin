  <behavior>
    statementPeriod({start, end}, rowDates):
    - stated start and end -> {from: start, toInclusive: end}
    - start only -> {from: start, toInclusive: max row date}; end only -> {from: min row date, toInclusive: end}
    - neither -> {from: min, toInclusive: max} over non-null row dates; no non-null date and no stated bound -> null
    - an inverted pair (from > to) -> null
    findExtras(pool, matchedIds, period, missing):
    - period null -> {extras: [], nearByIndex: empty}
    - a paid pool row inside the period and not in matchedIds is an extra; one in matchedIds is not
    - rows dated the day before `from` or the day after `toInclusive` are not extras (the ±2-day read margin is for matching only)
    - status 'pending' and 'skipped' rows are never extras
    - extras are ordered by localDate ascending, then id ascending
    - near-match: extra (−4200, 2026-03-05) and missing line {index 7, amount −4200, 2026-03-03} -> extra.nearIndex = 7 and nearByIndex.get(7) = extra.id; a 4-day gap -> no pairing; a different amount -> no pairing; a missing line with null date or null amount never pairs
    - each missing line pairs at most once: two extras of the same amount near one missing line -> only the earlier extra (by order above) pairs; with two candidate missing lines, the smaller day gap wins, then the smaller index
    - 5,000 pool rows and 5,000 missing lines complete in one call (test asserts the result length, not time)
    - property (fast-check): every returned extra id is in the pool, not in matchedIds, has status 'paid' and a date inside the period; no missing index appears twice in nearByIndex
  </behavior>

export function statementPeriod(stated: { start: string | null; end: string | null }, rowDates: readonly (string | null)[]): StatementPeriod | null;
