1. `create or replace function public.guard_plan_lock() returns trigger language plpgsql security definer set search_path = '' as $$` body:
   ```
   declare
     rec jsonb := to_jsonb(coalesce(new, old));
     hid uuid := nullif(rec->>'household_id','')::uuid;
     uid uuid := (select auth.uid());
   begin
     if uid is null or coalesce(current_setting('fincwin.plan_bypass', true), '') = 'on' then
       return coalesce(new, old);
     end if;
     if hid is null then
       select m.household_id into hid from public.household_members m where m.user_id = uid limit 1;
     end if;
     if hid is not null and not public.household_writable(hid) then
       raise exception 'plan locked' using errcode = 'PT402', hint = 'plan_locked';
     end if;
     return coalesce(new, old);
   end $$;
   ```
