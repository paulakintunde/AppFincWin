export type PlanPeriod = 'month' | 'year';

export interface PlanView {
  status: PlanStatus; kind: string; writable: boolean; householdId: string | null;
  planKey: PlanKey | null; planSize: number; period: PlanPeriod | null; seats: number; members: number;
