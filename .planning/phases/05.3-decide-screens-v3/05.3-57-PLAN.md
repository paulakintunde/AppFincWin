export type NotReadyAction =
  | { readonly kind: 'markPaid'; readonly lineId: string }

  readonly chips: readonly NotReadyChip[];
  readonly link: NotReadyChip | null;
}
export interface NotReadyGroup { readonly key: string; readonly month: YearMonth | null; readonly headingKey: string; readonly items: readonly NotReadyItemView[]; readonly monthLink: NotReadyChip | null }
