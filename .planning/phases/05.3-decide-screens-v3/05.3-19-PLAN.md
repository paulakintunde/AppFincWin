```ts
export function fundingStrip(ev: Evaluation): { headingKey: string; headingParams: Readonly<Record<string, number | string>>; parts: readonly StripPart[] } | null
export interface FactRow { kind: 'figure' | 'sentence' | 'coveredBy'; labelKey: string; labelParams: Readonly<Record<string, string | number>>; figure?: MinorUnits; negative?: boolean; subKey?: string; subParams?: Readonly<Record<string, string | number>>; tone?: 'warn' }
export function factRows(result: Result): readonly FactRow[]
export const OVER_RATE_TEXT = '999'; // D-11: payoff MAX_APR is 999.99% (pinned >= 999 by decideCopy.test.ts)
export function noticeCopy(n: Notice): { key: string; params: Readonly<Record<string, string | number>> }
export function factCopy(f: Fact): { key: string; params: Readonly<Record<string, string | number>> }
export function assumptions(ev: Pick<Evaluation, 'month' | 'result'>): { key: string; params: Readonly<Record<string, string | number>> }
```
