    - no question codes -> null; needsInputLines(['records_look_behind','no_price','account_use_question']) -> ['decide.v3.missing.records_look_behind','decide.v3.missing.no_price'] (question codes excluded, engine order kept)
    - askAnsweredProps(code) -> { missing: code } for the seven codes only
    - refundHint({ stored: {code:'not_yet'}, live: 'fits_unspent', without: 'not_yet' }) true; stored 'fits_unspent' -> false; live equals without -> false; without missing (no refund links) -> false; live worse than without -> false
    - changedSinceOpened('not_yet', 'not_covered', 'fits_unspent', null) -> { show: true, fromKey: chip of not_yet/not_covered, toKey: chip of fits_unspent }; same code and variant -> show false
    - changedSinceSaved(stored, liveResult) uses changeSinceLooked; stored null -> show false
  </behavior>

```ts
export const QUESTION_CODES: ReadonlySet<MissingCode> // the seven codes
export interface AskQuestion { code: MissingCode; questionKey: string; variant: 'fund' | null; params: Readonly<Record<string, string | number>>; pay: { seriesId: string; payDate: LocalDate; amount: MinorUnits } | null; months: readonly YearMonth[] }
export function askModel(ev: Pick<Evaluation, 'result' | 'month'>): { first: AskQuestion; k: number; n: number; more: number; moreKey: string | null } | null
export function needsInputLines(result: Pick<Result, 'missing'>): readonly string[]
/** Props every in-place answer body takes (05.3-32, 05.3-33). Type only. */
export interface AskBodyProps { question: AskQuestion; checkId: string; userId: string; householdId: string; onSaved(stepId: string | null): void }
export function askAnsweredProps(code: MissingCode): { missing: 'account_use_question' | 'pay_arrival_question' | 'unspent_not_confirmed' | 'unexplained_money_question' | 'income_pattern_unconfirmed' | 'expected_income_needed' | 'volatile_cover_question' }
export function refundHint(args: { stored: Pick<StoredSeenV3, 'code'> | null; live: Pick<Result, 'code'> | undefined; without: Pick<Result, 'code'> | undefined }): boolean
export function changedSinceOpened(fromCode: Code, fromVariant: Variant | null, toCode: Code, toVariant: Variant | null): { show: boolean; fromKey: string; toKey: string }
export function changedSinceSaved(stored: StoredSeenV3 | null, live: Result): { show: boolean; fromKey: string; toKey: string }
```
moreKey is null when more is 0. askAnsweredProps throws RangeError for a non-question code.
