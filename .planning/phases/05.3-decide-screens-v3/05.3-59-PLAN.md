```ts
// src/features/decide/v3/model/waiverNote.ts (pure)
export interface WaiverRow { readonly key: string; readonly textKey: string; readonly params: { name?: string; date?: LocalDate; month?: YearMonth }; readonly effectKey: string | null; readonly waiver: ReadinessWaiver; readonly name: string | null }
export function waiverRows(waived: readonly WaivedItem[] | undefined, paidByOther: MonthModel['paidByOther'] | undefined, alwaysOther: MonthModel['alwaysOther'] | undefined, thisMonth: YearMonth): readonly WaiverRow[];
export function waiverNote(rows: readonly WaiverRow[]): { readonly key: string; readonly params: Record<string, unknown> } | null; // null when no rows
```
- Rows come from waived (using WaivedItem.waiver), paidByOther and alwaysOther. Each alwaysOther entry becomes { always_other, seriesId } and takes its name from the entry. paidByOther entries become waivers as follows:
  - by 'paid_by_other' becomes { paid_by_other, seriesId, date };
  - by 'always_other' becomes { always_other, seriesId }.
- Deduplicate by waiver, using the kind plus its fields. One row per waiver: month_complete over three items is one row.
- effectKey is 'confidence.waiver.notCounted' for a paid_by_other row in thisMonth, and 'confidence.waiver.alwaysNotCounted' for every always_other row (D-31). Otherwise it is null.
- waiverNote:
  - one row gives { key: note_one, params: { text } }, where the card renders text from the row's textKey;
  - two or more rows give { key: note_other, params: { count } }.
