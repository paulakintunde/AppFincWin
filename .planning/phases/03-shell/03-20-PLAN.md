```ts
import type { QueryClient, QueryKey } from '@tanstack/react-query';
export type RefreshOutcome = 'updated' | 'nothingNew' | 'failed';
export async function runShellRefresh(qc: QueryClient, keys: readonly QueryKey[]): Promise<RefreshOutcome>;
export function useShellRefresh(keys: readonly QueryKey[]): { refreshing: boolean; refresh(): Promise<RefreshOutcome> };
```
`runShellRefresh`: `startedAt = Date.now()`; `before = keys.map((k) => qc.getQueryData(k))`; `try { await qc.resumePausedMutations(); } catch { /* failed writes stay with the sync line */ }`; `await Promise.all(keys.map((k) => qc.refetchQueries({ queryKey: k, type: 'active' })))`; `failed = keys.some((k) => { const s = qc.getQueryState(k); return s !== undefined && s.status === 'error' && s.errorUpdatedAt >= startedAt; })`; if failed return 'failed'; `changed = keys.some((k, i) => qc.getQueryData(k) !== before[i])`; `showToast({ kind: 'info', text: { key: changed ?
