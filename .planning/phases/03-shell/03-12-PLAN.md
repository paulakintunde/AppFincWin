```tsx
export function FabQuickMenu(): React.JSX.Element {
  const { quickMenuOpen, closeQuickMenu, quickActions } = useFab();
  const choose = (run: () => void) => { closeQuickMenu(); requestAnimationFrame(run); }; // close-then-open on the next frame (02.3 Pattern 4)
  return (
    <Sheet visible={quickMenuOpen} onDismiss={closeQuickMenu} accessibilityLabel={t('shell.quick.title')}>
      <SheetHeader title={t('shell.quick.title')} cancelLabel={t('record.sheet.cancel')} onCancel={closeQuickMenu} />
      {quickActions.map((a) => <Row key={a.id} label={a.label} onPress={() => choose(a.run)} />)}
    </Sheet>
  );
}
```
