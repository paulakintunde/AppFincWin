```ts
export function compactThreshold(largeHeaderHeight: number, insetsTop: number): number; // Math.max(0, largeHeaderHeight - (insetsTop + space.touchMin + 2 * space.gapSm))
export interface TabRootChromeOptions { title: string; eyebrow?: string; trailing?: React.ReactNode; refreshable?: boolean; refreshKeys?: readonly QueryKey[] }
export interface TabRootChrome {
  scrollRef: ReturnType<typeof useAnimatedRef>;
  onScroll: ReturnType<typeof useAnimatedScrollHandler>;
  scrollEventThrottle: 16;
  largeHeader: React.ReactElement;   // <View onLayout=measure><TabHeader title eyebrow trailing titleRef={largeTitleRef} /></View>
  compactHeader: React.ReactElement; // <CompactHeader title visible={compactVisible} animatedStyle={opacityStyle} titleRef={compactTitleRef} />
  refreshControl: React.ReactElement | undefined; // only when refreshable
}
export function useTabRootChrome(options: TabRootChromeOptions): TabRootChrome;
```
Implementation: `scrollY` shared value updated by `useAnimatedScrollHandler`; `threshold` shared value set from the large header's `onLayout` via `compactThreshold(height, insets.top)`; `useAnimatedReaction(() => scrollY.value > threshold.value, (now, prev) => { if (now !== prev) { opacity.value = withTiming(now ? 1 : 0, { duration: DURATION.fade }); runOnJS(setCompactVisible)(now); } })`; `opacityStyle = useAnimatedStyle(() => ({ opacity: opacity.value }))`. `useScrollToTop(scrollRef as never)` (D-08 second tap). Title focus: keep `focusRef = useRef<RNText | null>(null)`; an effect sets `focusRef.current = compactVisible ? compactTitleRef.current : largeTitleRef.current`; call `useScreenTitleFocus(focusRef)` (D-20). `refreshControl = refreshable ? <RefreshControl refreshing={refreshing} onRefresh={refresh} tintColor={colors.accent} colors={[colors.accent]} progressBackgroundColor={colors.surface} /> : undefined` (accent is reserved for exactly this, UI-SPEC Color). Reduced motion: the fade is a Reanimated timing, collapsed by the root ReducedMotionConfig (D-21).
