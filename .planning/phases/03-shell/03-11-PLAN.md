```ts
export function compactThreshold(largeHeaderHeight: number, insetsTop: number): number; // Math.max(0, largeHeaderHeight - (insetsTop + space.touchMin + 2 * space.gapSm))
export interface TabRootChromeOptions { title: string; eyebrow?: string; trailing?: React.ReactNode; refreshKeys?: readonly QueryKey[] }
export interface TabRootChrome {
  scrollRef: ReturnType<typeof useAnimatedRef>;
  onScroll: ReturnType<typeof useAnimatedScrollHandler>;
  scrollEventThrottle: 16;
  largeHeader: React.ReactElement;   // <View onLayout=measure><TabHeader title eyebrow trailing titleRef={largeTitleRef} /></View>
  compactHeader: React.ReactElement; // <CompactHeader title visible={compactVisible} animatedStyle={opacityStyle} titleRef={compactTitleRef} />
  refreshControl: React.ReactElement | undefined;
}
export function useTabRootChrome(options: TabRootChromeOptions): TabRootChrome;
```
Implementation: `scrollY` shared value from `useAnimatedScrollHandler`; `threshold` set from the large header's `onLayout` via `compactThreshold(height, insets.top)`; `useAnimatedReaction(() => scrollY.value > threshold.value, (now, prev) => { if (now !== prev) { opacity.value = withTiming(now ? 1 : 0, { duration: DURATION.fade }); runOnJS(setCompactVisible)(now); } })`. `useScrollToTop(scrollRef as never)`. Focus: `focusRef` follows the visible title; `useScreenTitleFocus(focusRef)`. Refresh: `const { refreshing, refresh } = useShellRefresh(options.refreshKeys ?? [])`; `onRefresh = async () => { await refresh(); if (await AccessibilityInfo.isScreenReaderEnabled()) focusAccessibility(focusRef.current); }`; `refreshControl = options.refreshKeys ? <RefreshControl refreshing={refreshing} onRefresh={onRefresh} tintColor={colors.accent} colors={[colors.accent]} progressBackgroundColor={colors.surface} /> : undefined` (accent is reserved for this, UI-SPEC Color).
