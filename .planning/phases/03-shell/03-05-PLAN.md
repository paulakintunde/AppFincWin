1. `src/theme/fontScale.ts` (header: "DSG-07 / D-19: body text and labels follow the OS setting to 200% (WCAG 1.4.4); large display figures (net worth 42px, health score 32px) and tab labels cap at about 1.3x so the design survives. Applied manually from the OS font scale rather than via maxFontSizeMultiplier, which is reported ignored on Fabric (facebook/react-native#47499, #35658; 03-RESEARCH Pitfall 2)."):
   ```ts
   import type { TextStyle } from 'react-native';
   export const FONT_SCALE = { body: 2, capped: 1.3 } as const;
   export type FontScaleTier = keyof typeof FONT_SCALE;
   export function effectiveScale(osScale: number, tier: FontScaleTier): number; // !(osScale > 0) or not finite -> 1; else Math.min(osScale, FONT_SCALE[tier])
   export function scaleTextStyle(style: Pick<TextStyle, 'fontSize' | 'lineHeight' | 'letterSpacing'>, scale: number): Pick<TextStyle, 'fontSize' | 'lineHeight' | 'letterSpacing'>; // multiplies only present numeric props; scale === 1 -> {}
   ```
2. `src/theme/FontScaleProvider.tsx`: `const FontScaleContext = createContext<number>(1);` `export function FontScaleProvider({ children, value }: { children: React.ReactNode; value?: number })` — uses `useWindowDimensions().fontScale` when `value` is undefined (the `value` prop exists for tests); `export function useFontScale(): number` returns the context (1 outside a provider).
3. `src/ui/Text.tsx`:
   ```tsx
   import React, { forwardRef } from 'react';
   import { StyleSheet, Text as RNText, type TextProps as RNTextProps, type TextStyle } from 'react-native';
   import { effectiveScale, scaleTextStyle, type FontScaleTier } from '@/theme/fontScale';
   import { useFontScale } from '@/theme/FontScaleProvider';
   export interface TextProps extends RNTextProps { /** D-19 tier: 'body' (default) up to 2x, 'capped' up to 1.3x for display figures and tab labels. */ scale?: FontScaleTier }
   export const Text = forwardRef<RNText, TextProps>(function Text({ scale = 'body', style, allowFontScaling, ...rest }, ref) {
     const os = useFontScale();
     if (allowFontScaling === false) return <RNText ref={ref} {...rest} allowFontScaling={false} style={style} />;
     const flat = (StyleSheet.flatten(style) ?? {}) as TextStyle;
     const scaled = scaleTextStyle(flat, effectiveScale(os, scale));
     return <RNText ref={ref} {...rest} allowFontScaling={false} style={[style, scaled]} />;
   });
   ```
   (`allowFontScaling={false}` stops the OS scaling on top of the manual scale; the manual scale is the only scaling.)
