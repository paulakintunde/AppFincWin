import type { Location } from './backHistoryReducer';
export const OUTER_GROUP = '(app)';

Algorithm: walk focused routes (`routes[index ?? routes.length - 1]`) from the root until a route named OUTER_GROUP; if none or it has no `state`, return null. outer = that route's state; oi = outer.index ?? last; focused = outer.routes[oi]. (stack.index ?? stack.routes.length - 1) : 0; return `{ area: 'tab', tab: tabRoute.name, depth, href: pathname }`.
