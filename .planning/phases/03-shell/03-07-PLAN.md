```ts
export interface FabConfig { label: string; accessibilityLabel: string; onPress(): void }
export interface QuickActionEntry { id: QuickActionId; label: string; run(): void }
export interface FabContextValue {
  config: FabConfig | null;
  hidden: boolean;
  quickActions: readonly QuickActionEntry[];
  quickMenuOpen: boolean;
  openQuickMenu(): void;
  closeQuickMenu(): void;
  setScreenFab(config: FabConfig | null): void;
  setDefaultAction(fn: (() => void) | null): void;
  setQuickAction(id: QuickActionId, fn: (() => void) | null): void;
  holdHide(reason: FabHideReason): () => void; // returns a release function; reasons are reference-counted
}
export function FabProvider({ children }: { children: React.ReactNode }): React.JSX.Element;
export function useFab(): FabContextValue; // throws 'useFab must be used within a FabProvider'
export function useRegisterFab(config: FabConfig | null): void;           // useFocusEffect: set on focus, setScreenFab(null) on blur; keeps onPress in a ref so the effect deps are [label, accessibilityLabel] only
export function useProvideDefaultFabAction(fn: (() => void) | null): void; // useEffect: set on mount/change, null on unmount; onPress via ref
export function useProvideQuickAction(id: QuickActionId, fn: (() => void) | null): void;
export function useHideFab(reason: FabHideReason, active: boolean): void;  // useEffect(() => active ? holdHide(reason) : undefined, [reason, active])
```
