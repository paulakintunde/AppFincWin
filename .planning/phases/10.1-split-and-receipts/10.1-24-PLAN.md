```ts
export interface ReviewItem { key: string; name: string; price: number | null; discount: number; categoryId: string | null; likely: boolean; unsure: boolean; frame: ReceiptFrame | null; page: number }
export interface ReviewExtra { key: string; kind: 'tax' | 'tip' | 'discount'; name: string; amount: number | null; unsure: boolean }
export interface ReviewDraft { mode: 'new' | 'existing'; store: Field<string | null>; date: Field<string | null>; total: Field<number | null>; items: ReviewItem[]; extras: ReviewExtra[]; totalChoice: 'receipt' | 'items' | null; taxTipCategoryId: string | null; images: { uri: string; mime: string }[] }
export function fromParsed(o: ParseOutcome | null, ctx: { mode; images; lineTotal?: number; lineCategoryId?: string | null; suggest(name: string, store: string | null): { categoryId: string | null; source: string } }): ReviewDraft;   // existing mode: total = lineTotal (sure); null outcome or unreadable -> empty items
export function reviewBalance(d: ReviewDraft): Reconcile;
export function canConfirm(d: ReviewDraft): { ok: true } | { ok: false; reason: 'noItems' | 'emptyPrice' | 'noTotal' | 'mismatch' };
export function useReceiptTotal(d: ReviewDraft): ReviewDraft;   // gap > 0 -> add item 'Other' (scan.review.other) with price gap; gap < 0 -> add discount extra 'Other' of |gap|; totalChoice 'receipt'
export function useItemsTotal(d: ReviewDraft): ReviewDraft;     // total := computed; totalChoice 'items'
export function toPrivateSave(d: ReviewDraft, lineSign: 1 | -1, lineCategoryId: string | null): { total: number; parts: Part[] | null; categoryId: string | null };   // via itemsToParts; categoryId = the single category when parts null
```
(These helpers are plain functions despite the `use` prefix in two names; name them `applyReceiptTotal` and `applyItemsTotal` to satisfy the React hooks lint rule.)
