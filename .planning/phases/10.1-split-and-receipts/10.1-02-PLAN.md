```ts
// src/services/receipts/types.ts
export type ReceiptMime = 'image/jpeg' | 'image/png' | 'image/webp' | 'application/pdf';
export type ReceiptKind = 'photo' | 'file' | 'scan';
export interface LocalReceiptFile { uri: string; mime: ReceiptMime; ext: 'jpg' | 'png' | 'webp' | 'pdf'; bytes: number; width: number | null; height: number | null }
export type PickResult =
  | { ok: true; kind: ReceiptKind; files: LocalReceiptFile[] }   // scan: one file per page, in page order
  | { ok: false; reason: 'cancelled' | 'permission_denied' | 'too_big' | 'bad_type' | 'unavailable' | 'failed' };
export interface OcrFrame { left: number; top: number; width: number; height: number }
export interface OcrLine { text: string; frame: OcrFrame | null }   // frame null when the module returns text only
export interface OcrResult { lines: OcrLine[]; imageWidth: number | null; imageHeight: number | null }
export type ReadResult = { ok: true; result: OcrResult } | { ok: false; reason: 'unavailable' | 'failed' | 'empty' };
// src/services/receipts/limits.ts
export const RECEIPT_MAX_FILES_PER_LINE = 5;      // D-21 (a multi-page scan counts as one receipt)
export const RECEIPT_MAX_PAGES_PER_SCAN = 10;
export const RECEIPT_MAX_BYTES = 5 * 1024 * 1024; // 5242880, D-21, matches the bucket file_size_limit
export const RECEIPT_LONG_EDGE_PX = 2000;          // D-21
export const RECEIPT_JPEG_QUALITY = 0.8;
```
Service signatures:
pickPhoto(source: 'camera' | 'library', deps?) -> Promise<PickResult>   (expo-image-picker launchCameraAsync / launchImageLibraryAsync, mediaTypes ['images'], allowsMultipleSelection false, exif false, quality 1; result passes through prepareImage)
pickFile(deps?) -> Promise<PickResult>   (expo-document-picker getDocumentAsync({ type: ['image/*','application/pdf'], copyToCacheDirectory: true, multiple: false }); PDFs pass through unchanged if <= RECEIPT_MAX_BYTES; images through prepareImage)
scanDocument(deps?) -> Promise<PickResult>   (scanner module, pageLimit RECEIPT_MAX_PAGES_PER_SCAN, JPEG pages, each page through prepareImage; kind 'scan')
prepareImage(uri, deps?) -> Promise<{ ok: true; file: LocalReceiptFile } | { ok: false; reason: 'too_big' | 'bad_type' | 'failed' }>   (expo-image-manipulator: resize so the long edge is at most 2000 px (never upscale), save JPEG at quality 0.8, so HEIC/PNG become JPEG; then size check)
readOnDevice(imageUri, deps?) -> Promise<ReadResult>
</interfaces>
