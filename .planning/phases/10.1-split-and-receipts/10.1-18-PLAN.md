```ts
export interface DraftPart { key: string; categoryId: string | null; amount: number | null; touched: boolean }   // amount: positive magnitude minor units; null = empty field
export interface SplitDraft { total: number /* positive magnitude */; sign: 1 | -1; parts: DraftPart[]; firstIsRest: boolean }
export function initDraft(line: { amount: number; categoryId: string | null; parts: Part[] | null }): SplitDraft;   // existing parts -> prefill (firstIsRest false); else [line category with total, empty] (firstIsRest true)
export function setAmount(d: SplitDraft, key: string, amount: number | null): SplitDraft;   // editing part 1 sets firstIsRest false; while firstIsRest, part 1 = max(0, total - sum(others))
export function setCategory(d: SplitDraft, key: string, categoryId: string | null): SplitDraft;
export function addPart(d: SplitDraft): SplitDraft;   // max 20 parts
export function removePart(d: SplitDraft, key: string): SplitDraft;   // no-op at 2 parts
export function draftBalance(d: SplitDraft): Balance;
export function canSave(d: SplitDraft): { ok: true } | { ok: false; reason: 'needsAmount' | 'notAddingUp' | 'tooFew' };
export function toSaveParts(d: SplitDraft): Part[];   // signed with d.sign, order kept
```
