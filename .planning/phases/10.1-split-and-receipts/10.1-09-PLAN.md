```ts
import type { V3Line } from '../decide/v3/types';
export function linePartAmounts(line: Pick<V3Line, 'id' | 'amount' | 'categoryId' | 'parts'>): { categoryId: string | null; amount: number }[];
// no parts (absent or < 2) -> [{ categoryId: line.categoryId, amount: line.amount }]
// parts -> expandParts({ id, amount: line.amount, categoryId, parts }) mapped to { categoryId, amount } (gap row categoryId null)
export function scaleLineParts<L extends Pick<V3Line, 'id' | 'amount' | 'categoryId' | 'parts'>>(line: L, newAmount: number): L['parts'];  // parts re-spread over newAmount with scaleParts; undefined when the line has no parts
// src/engine/parts/rowParts.ts  (for Activity, caps and the Decide builder; plain data in, no db types)
export function homeCategoryAmounts(line: LinePartsInput, amountHome: number | null): CategoryAmount[] | null;  // null when amountHome is null; unsplit -> one row with amountHome; split -> scaleParts(line, amountHome)
export function partsMatch(line: LinePartsInput, categoryIds: readonly (string | null)[]): boolean;            // true when the line's category or any part's category (gap = null) is in the list
export function amountInCategories(line: LinePartsInput, amountHome: number | null, categoryIds: readonly (string | null)[]): number | null;  // sum of the matching home-scaled rows; null when amountHome is null
```
