<objective>
Make receipts offline-safe and private on the phone side: a persisted upload outbox with its own cap, an idempotent uploader (upload then add_receipt), short-lived signed links for viewing, PDF hand-off to the system viewer, and a host that drains the outbox.

Purpose: D-09 "nothing is lost"; D-07 access stays server-enforced while signed links never land in the persisted cache. Receipt uploads are not row writes, so they do not consume Phase 10's D-27 limit (planner decision recorded here).
Output: outbox, upload and open services with tests; outbox host mounted in the app layout.

Phase 10 dependency: Phase 10 changed the write queue and wipe flows (plans 10-12, 10-13). Read the merged src/services/storage/wipe.ts and src/data/sync/* before wiring; if Phase 10 is not merged, stop and report.

```ts
// src/services/receipts/outbox.ts
export const OUTBOX_KEY = 'fincwin:receiptOutbox';
export const OUTBOX_DIR = 'receipts-pending/';     // under FileSystem.documentDirectory
export const OUTBOX_MAX_FILES = 40;
export const OUTBOX_MAX_BYTES = 150 * 1024 * 1024;
export const MAX_ATTEMPTS = 5;
export interface OutboxItem { receiptId: string; groupId: string; pageNo: number; transactionId: string; kind: ReceiptKind; mime: ReceiptMime; ext: string; bytes: number; localUri: string; attempts: number; nextAttemptAt: string | null; state: 'waiting' | 'failed'; createdAt: string }
export interface OutboxDeps { readList(): Promise<OutboxItem[]>; writeList(items: OutboxItem[]): Promise<void>; copyIn(src: string, dest: string): Promise<void>; remove(uri: string): Promise<void>; now(): Date; newId(): string; isOnline(): boolean; uploadOne(item: OutboxItem): Promise<UploadOutcome> }
export type EnqueueResult = { ok: true; items: OutboxItem[] } | { ok: false; reason: 'outbox_full' | 'too_big' };
export function enqueueReceipt(input: { transactionId: string; kind: ReceiptKind; files: LocalReceiptFile[] }, deps?: Partial<OutboxDeps>): Promise<EnqueueResult>;  // one group id for all pages; copies files into OUTBOX_DIR/<receiptId>.<ext>
export function drainOutbox(deps?: Partial<OutboxDeps>): Promise<{ uploaded: number; waiting: number; failed: number }>;  // single-flight; skips items whose nextAttemptAt is in the future
export function listOutbox(transactionId?: string): Promise<OutboxItem[]>;
export function removeFromOutbox(groupId: string): Promise<void>;   // deletes local files
export function retryOutboxItem(groupId: string): Promise<void>;     // failed -> waiting, attempts 0
// src/services/receipts/upload.ts
export type UploadOutcome = 'uploaded' | 'line_not_ready' | 'retry' | 'rejected';
export function uploadOne(item: OutboxItem, deps?: Partial<UploadDeps>): Promise<UploadOutcome>;
//  path := receiptObjectPath(...) (null -> 'line_not_ready'); storage.from('receipts').upload(path, bytes, { contentType: mime, upsert: false }) (409 counts as uploaded);
//  addReceipt(...): ok -> 'uploaded'; object_missing / network -> 'retry'; too_many_receipts / too_many_pages / too_big / bad_type / not_found -> 'rejected'
// src/services/receipts/openReceipt.ts
export const SIGNED_URL_SECONDS = 60;
export function signedUrlFor(objectPath: string, deps?): Promise<string | null>;
export function openPdf(objectPath: string, deps?): Promise<'opened' | 'failed'>;   // downloads to cacheDirectory/receipt-view/<file>, opens with expo-sharing shareAsync({ mimeType: 'application/pdf', UTI: 'com.adobe.pdf' }), deletes the temp file on the next call
```
Backoff: attempt n waits min(2^n minutes, 60 minutes); 'line_not_ready' does not count as an attempt; 'rejected' moves the item to failed immediately; after MAX_ATTEMPTS 'retry' outcomes the item is failed.
