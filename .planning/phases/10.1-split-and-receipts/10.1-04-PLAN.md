```ts
// src/engine/parts/types.ts
export interface Part { readonly categoryId: string | null; readonly amount: number; readonly label?: 'taxTip' | null }   // signed minor units, same sign as the line
export interface LinePartsInput { readonly id: string; readonly amount: number; readonly categoryId: string | null; readonly parts: readonly Part[] | null }
export interface CategoryAmount { readonly lineId: string; readonly categoryId: string | null; readonly amount: number; readonly isGap: boolean; readonly label: 'taxTip' | null }
export type PartsInvalidReason = 'too_few' | 'too_many' | 'zero_part' | 'mixed_sign' | 'not_integer' | 'over_total' | 'not_total';
export type Balance = { readonly state: 'addsUp'; readonly amount: 0 } | { readonly state: 'left' | 'over'; readonly amount: number };  // amount is a positive magnitude
// expandParts.ts
export function expandParts(line: LinePartsInput): CategoryAmount[];
// validateParts.ts
export const PARTS_MIN = 2; export const PARTS_MAX = 20;
export function validateParts(total: number, parts: readonly Part[], opts?: { allowGap?: boolean }): PartsInvalidReason | null;  // allowGap default false: the sum must equal the total (saving from the sheet); true: abs(sum) <= abs(total) (a stored line after a total edit)
// partsBalance.ts
export function partsBalance(total: number, amounts: readonly (number | null)[]): Balance;   // null = empty field, counts as 0
// primaryCategory.ts
export function primaryCategory(parts: readonly Part[]): string | null;   // largest abs amount, ties to the lowest index
// scaleParts.ts
export function scaleParts(line: LinePartsInput, target: number): CategoryAmount[];   // amounts sum exactly to target
// src/engine/split/allocateWithZeros.ts
export function allocateWithZeros(total: MinorUnits, weights: readonly number[]): MinorUnits[];
```
Rules: 
- expandParts: parts null or length < 2 -> `[{ lineId, categoryId: line.categoryId, amount: line.amount, isGap: false, label: null }]`; otherwise one row per part (label kept) and, when `gap = amount - sum(parts) !== 0`, one more row `{ categoryId: null, amount: gap, isGap: true, label: null }` (the gap may have either sign: "over" after a total edit gives a gap of the opposite sign, D-03).
- scaleParts: unsplit -> one row with `target`; split -> weights are `abs(part.amount)` for each part plus `abs(gap)` when the gap has the line's sign (left to place); when the gap is "over" (opposite sign) the gap is dropped and the target is spread over the parts only; zero-weight entries get 0; result rows keep categoryId/label/isGap and sum to `target`. Uses allocateWithZeros(target, weights). If every weight is 0 (amount 0 line), return the expandParts rows with amount 0 except the first which gets `target`.
- allocateWithZeros: indexes with weight 0 get 0; positive weights go through allocate unchanged; all weights 0 throws RangeError('allocateWithZeros: all weights are zero') (mirrors SQL 22023); negative weight throws RangeError.
