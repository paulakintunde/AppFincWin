```ts
// amounts.ts
export function parsePriceToMinor(token: string, exponent: number): number | null;   // RESEARCH "Parsing a price into minor units without floats"; null = not a price or ambiguous grouping
export function rowPrice(text: string, exponent: number): { amount: number; negative: boolean; token: string; ambiguous: boolean } | null;  // rightmost price-like token
// dates.ts
export type DateOrder = 'MDY' | 'DMY' | 'YMD';
export function findDate(texts: readonly string[], order: DateOrder): Field<string | null>;   // ISO YYYY-MM-DD; unsure when both orders are valid and differ
// keywords.ts
export type RowKind = 'total' | 'subtotal' | 'tax' | 'tip' | 'discount' | 'payment' | 'change' | 'item';
export function classifyRow(text: string): RowKind;   // total: total, amount due, balance due, grand total, to pay; tax: tax, gst, hst, pst, qst, vat, sales tax; tip: tip, gratuity, service charge; discount: discount, savings, coupon, promo, you saved; payment: cash, card, visa, mastercard, amex, debit, credit, tendered, auth; change: change, change due
// rows.ts
export interface OcrRow { text: string; frame: ReceiptFrame | null; lineIndexes: number[] }
export function groupRows(ocr: ReceiptOcr): OcrRow[];   // frames: group by vertical centre within half the median line height, order rows top-down and words left-right; no frames: one row per line in module order
// parseReceipt.ts
export interface ParseOptions { currency: string; dateOrder: DateOrder; today: string }
export type ParseOutcome = { readable: true; receipt: ParsedReceipt } | { readable: false; receipt: ParsedReceipt };   // unreadable = no items and no total; receipt is then the empty shell (store/date/total null, unsure true)
export function parseReceipt(ocr: ReceiptOcr, opts: ParseOptions): ParseOutcome;
// reconcile.ts
export function reconcile(r: Pick<ParsedReceipt, 'items' | 'extras' | 'total'>): Reconcile;   // computed = sum(price - discount) + tax + tip - discounts
```
Parser rules: rows after the total row are ignored except tip rows on restaurant receipts; 'payment' and 'change' rows are dropped; 'subtotal' is used only as a cross-check (if items sum != subtotal, flag every item whose price token was ambiguous, else flag the subtotal mismatch on the total); a negative or "-" price row directly under an item becomes that item's discount (ExtraKind not used); other discount rows become receipt-level 'discount' extras; a row without a price directly under an item is a continuation of the item name; "2 x 1.50" / "2 @ 1.50" quantity rows multiply into the item price exactly (integer maths); the store is the first row in the top fifth (or first 3 rows without frames) that has letters, no price and is not a date, phone or address-number row, flagged unsure unless it is the very first row; total missing -> total {value: null, unsure: true}; item names capped at 120 characters; item keys `i<index>`, extra keys `x<index>`.
