    - cover (max 30): e = max(emergency ?? 0, 0); month = monthlyEssential > 0 ? monthlyEssential : moneyOut; points = divideHalfUp(30n * min(e, 3n*month), 3n*month)
    - debt (max 20): base = typicalIncome > 0 ? typicalIncome : moneyIn; yearly = 12n*base; d = max(debtTotal,0); points = divideHalfUp(20n * (3n*yearly - min(5n*d, 3n*yearly)), 3n*yearly)
    - billsToPay (max 15, D-20 "Bills still to pay this month"): points = divideHalfUp(15n * (3n*moneyOut - min(5n*billsDue, 3n*moneyOut)), 3n*moneyOut) with billsDue clamped at 0

    Display figures (integers, never floats): kept keptPercent = posLeft === 0 ? 0 : divideHalfUp(100n*posLeft, moneyIn); cover coverTenths = divideHalfUp(10n*e, month); debt debtPercent = divideHalfUp(100n*d, yearly); billsToPay billsPercent = divideHalfUp(100n*billsDue, moneyOut).
    Thin data (D-05), checked in this order: moneyIn <= 0 -> {kind:'thin', missing:'income'}; moneyOut <= 0 -> 'spending'; hasBalance false -> 'balance'.

    import type { Prepared } from '../decide/v3/prepare';
    import { minorFromBigInt } from '../payoff/types';
    const SHORT_CODES = new Set(['typical_short', 'month_heading_short']);
    export function healthInputsOf(p: Pick<Prepared, 'inputs' | 'readings'>): HealthInputs {
      const r = homeReading(p); const pic = monthPicture(p);
      let debt = 0n; for (const d of r.inputs.debts) if (d.balance > 0) debt += BigInt(d.balance);
      return { moneyIn: r.month.moneyIn, left: pic.leftAfterPlans, emergency: r.funds.emergency, monthlyEssential: r.net.starter,
        debtTotal: minorFromBigInt(debt, FN), typicalIncome: r.month.typicalIncome, billsDue: r.month.billsDue, moneyOut: r.month.moneyOut,
        hasBalance: (r.inputs.accounts?.length ?? 0) > 0,
        short: pic.leftAfterPlans < 0 || r.month.notices.some((n) => SHORT_CODES.has(n.code)) };
