```tsx
export interface DisplayChipProps { view: ActivityView; sort: SortKey; onPress: () => void; chipRef?: React.Ref<View> }
export function DisplayChip({ view, sort, onPress, chipRef }: DisplayChipProps) {
  const t = useT();
  const viewName = t(`activity.view.${view}`);
  const calendar = view === 'calendar';
  const label = calendar ? viewName : t('activity.display.chip', { view: viewName, sort: t(`activity.sort.${sort}`) });
  const a11y = calendar ? t('activity.display.triggerA11yCalendar') : t('activity.display.triggerA11y', { view: viewName, sort: t(`activity.sort.${sort}`) });
  return <Chip label={label} accessibilityLabel={a11y} onPress={onPress} pressableRef={chipRef} />;
}
```
