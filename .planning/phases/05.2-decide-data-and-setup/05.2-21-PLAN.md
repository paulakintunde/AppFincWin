// setupFlow.ts (pure; no React)
export const SETUP_STEP_KEYS = ['welcome','paid','pay','season','method','other','regular','loans','saving','everyday','unusual','accountUse','emergency','envelopes','unspent','share','detail','reminders','summary'] as const;
export type SetupStepKey = (typeof SETUP_STEP_KEYS)[number];
export type SetupSection = 'income' | 'regular' | 'spending' | 'money' | 'answers';
export interface SetupFacts {
  incomeTypes: readonly string[] | null;          // decide_settings.income_types (6 setup values)
  hasRegularPaySeries: boolean;                    // a live income series of kind salary exists
  method: 'match'|'spread'|'steady'|'balance'|null;
  spreadMonths: number | null; steadyAmount: number | null; holdMonths: number | null;
  fullMonthsOfRecords: number;
  moneyAccountIds: readonly string[];              // accounts that hold money (not credit, not loan)
  accountIdsWithUse: readonly string[];            // stored use non-null
  everydayOverride: number | null;
  unspentAnchorSet: boolean;
  roughTyped: boolean;                             // any rough_* figure typed with < 3 months of records
  setupDoneAt: string | null;
}
export function sectionOf(step: SetupStepKey): SetupSection | null;     // welcome/summary -> null
export function isNeeded(step: SetupStepKey, facts: SetupFacts): boolean;
export function visibleSteps(facts: SetupFacts, mode: 'full' | 'standalone'): SetupStepKey[];
export function isAnswered(step: SetupStepKey, facts: SetupFacts): boolean;
export function setupStatus(facts: SetupFacts): 'not_started' | 'rough' | 'done';
export function firstUnansweredNeeded(facts: SetupFacts): SetupStepKey | null;
export function remainingNeeded(facts: SetupFacts): SetupStepKey[];
// steps/types.ts
export interface SetupStepBodyProps {
  readonly mode: 'stepper' | 'edit' | 'standalone';
  readonly onReadyChange: (ready: { canContinue: boolean; reasonKey: string | null; dirty: boolean }) => void;
}
export interface SetupStepHandle {
  /** Writes the step's answers through its mutations. Returns the undo step id, or null when nothing changed (no write, no toast). */
  save(): { stepId: string | null };
}
