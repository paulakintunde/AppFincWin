export type RefineTrigger = 'three_months' | 'rough_one_month' | 'typed_off' | 'new_repeating' | 'life_change';
export type RefineCardKind =
  | 'pay' | 'pay_stopped' | 'everyday' | 'everyday_override_expiring' | 'new_regular' | 'saving_bucket'
  | 'unspent' | 'changing_income' | 'steady_above_income' | 'series_drift' | 'card_not_paid_in_full' | 'under_logged_month';
export interface RefineCard {
  readonly kind: RefineCardKind;
  readonly ref: string | null;            // series id, debt id, envelope id, candidate key or null
  readonly month: YearMonth | null;       // under_logged_month only
  readonly current: MinorUnits | null;    // what Decide uses now
  readonly suggested: MinorUnits | null;  // what the records show
  readonly noise: MinorUnits | null;      // the noise the change beat (value cards only)
}
export interface RefineInput {
  readonly inputs: V3Inputs;                       // records + stored answers (asOf, months, series, profile, everydayOverride, unspent, envelopes, debts)
  readonly setupDoneAt: LocalDate | null;          // date part of decide_settings.setup_done_at
  readonly setupRough: boolean;                    // setup finished on rough figures
  readonly overrideAt: LocalDate | null;           // decide_settings.everyday_override_at
  readonly lastAt: LocalDate | null;               // date part of refine_last_at
  readonly snoozeUntil: LocalDate | null;          // date part of refine_snooze_until
  readonly candidates: readonly { readonly key: string; readonly occurrences: number; readonly amount: MinorUnits }[]; // from detectCandidates, unmarked only
  readonly cardStatements: readonly { readonly debtId: string; readonly statedPaysInFull: 'yes' | 'sometimes' | 'no' | null; readonly paidInFull: readonly boolean[] }[]; // newest last
}
export interface RefineResult { readonly due: boolean; readonly triggers: readonly RefineTrigger[]; readonly cards: readonly RefineCard[] }
export function refine(input: RefineInput): RefineResult;
export function refineNoise(current: MinorUnits, window: readonly MinorUnits[]): MinorUnits;

  <context>
No noise or spread helper exists in the engine (confidence.ts has none). Proposed rule, all integer minor units, all tagged "FincWin choice" in constants.ts:

Noise for one figure = the largest of:
  (a) REFINE_NOISE_MIN_MINOR = 2,500 (that is $25 in a 2-decimal currency);
  (b) REFINE_NOISE_PCT = 5% of the current figure (rounded up);
  (c) the median absolute deviation of that figure across the last REFINE_WINDOW_MONTHS = 6 full months (0 when fewer than 3 months exist).
A value card shows only when |suggested - current| > noise.

Triggers and other numbers (from 00-UPGRADE-PLAN §4.12/§4.5 and D-20 where given; FincWin choice where marked):
- REFINE_SETUP_MONTHS = 3 full months since setup; REFINE_ROUGH_MONTHS = 1 full month after a rough setup
- REFINE_TYPED_OFF_PCT = 15, REFINE_TYPED_OFF_MONTHS = 2 (a typed figure 15% off the records two months running)
- REFINE_REPEAT_COUNT = 3 (new regular payment or income repeating 3 times)
- REFINE_MIN_GAP_DAYS = 30 (at most once a month, FincWin choice for "a month"); REFINE_SNOOZE_DAYS = 14
- REFINE_OVERRIDE_MAX_DAYS = 90; REFINE_OVERRIDE_MEDIAN_PCT = 20
- REFINE_STEADY_ABOVE_MONTHS = 2; REFINE_CARD_UNPAID_STATEMENTS = 2
- REFINE_UNDERLOGGED_PCT = 50 (FincWin choice: a full month whose everyday spending is under 50% of the typical month is suggested once as under-logged)
- Series drift reuses the existing seriesDrift / SERIES_DRIFT_DIVISOR = 20 (5%)
- REFINE_PAY_STOPPED_MISSES = 2 (FincWin choice: a regular pay with no matched paid line for its last 2 expected dates)
