```typescript
export type CarryClamp = 'none' | 'cap' | 'floor';
export interface CarryMonth {
  month: string;            // YYYY-MM
  limit: number;            // plain limit L
  effective: number;        // E, 0..2L
  carried: number;          // E - L, -L..L
  spent: number;            // this month's net spend from spentByMonth (missing = 0)
  clamp: CarryClamp;        // how E was clamped from the previous month ('none' for the first month)
  prevRemaining: number | null; // R of the previous month (E_prev - s_prev); null for the first month
}
export function isYearMonth(value: string): boolean; // /^\d{4}-(0[1-9]|1[0-2])$/
export function chainStartMonth(rolloverSince: string | null, rolloverFrom: string | null): string | null;
//   null when rolloverSince is null; else the later of rolloverSince and (rolloverFrom ?? rolloverSince) by string compare.
//   Throws RangeError if a non-null argument fails isYearMonth.
export function carryChain(input: {
  limit: number; startMonth: string; throughMonth: string;
  spentByMonth: Readonly<Record<string, number>>;
}): readonly CarryMonth[];
//   RangeError unless Number.isSafeInteger(limit) && limit >= 1, both months pass isYearMonth, every value in spentByMonth is a safe integer.
//   [] when startMonth > throughMonth. First entry: effective = limit, carried 0, clamp 'none', prevRemaining null.
//   Next: R = prev.effective - prev.spent; raw = limit + R; effective = Math.min(2 * limit, Math.max(0, raw));
//   clamp = raw > 2 * limit ? 'cap' : raw < 0 ? 'floor' : 'none'; carried = effective - limit; prevRemaining = R.
export function effectiveIn(chain: readonly CarryMonth[], month: string, limit: number): number;
//   the chain entry's effective for `month`, else `limit` (months before the chain start or after it are plain).
```
