```typescript
export interface DuplicateCandidate { index: number; localDate: string | null; amount: number | null; name: string; externalId: string | null }
export interface ExistingRow { id: string; localDate: string; amount: number; name: string | null; externalId: string | null; importFormat: string | null }
export type DuplicateMatch = { kind: 'existing'; id: string; by: 'fitid' | 'match' | 'cross-format-window' };
export const NAME_SIMILARITY_THRESHOLD = 0.6;
export const CROSS_FORMAT_WINDOW_DAYS = 2;
export function nameSimilarity(a: string, b: string): number; // Jaccard over normalised word tokens; 1 when either normalised string contains the other; an existing null name counts as 1 (pre-Record rows had no name)
export function fitidReliable(candidates: readonly DuplicateCandidate[]): boolean; // false when one FITID appears on rows with different amounts or dates
export function findDuplicates(candidates: readonly DuplicateCandidate[], existing: readonly ExistingRow[], opts: { sourceFormat: ImportSource }): { matches: Map<number, DuplicateMatch>; fitidDisabled: boolean };
```
Algorithm:
1. `fitidDisabled = !fitidReliable(candidates)`.
2. FITID pass (skipped when disabled): for each candidate with externalId, find an unused existing row with the same externalId AND the same amount (date may differ) -> flag `by: 'fitid'`, mark the existing row used.
3. Multiset pass for the remaining candidates: group remaining existing rows by `amount`; for each candidate (in index order) consider unused existing rows with the same amount whose date is equal, OR — only when `existing.importFormat !== null && existing.importFormat !== opts.sourceFormat` — within ±CROSS_FORMAT_WINDOW_DAYS days (date difference via Date.UTC on the parts). Among those with nameSimilarity >= threshold pick the best (similarity desc, exact date first, smaller day gap, id asc); flag it (`by: 'match'` for the exact date, `'cross-format-window'` otherwise) and mark the existing row used. Candidates never compare with each other.
4. Candidates with null date or amount are skipped.
