```typescript
export interface KeywordRule { keyword: string; category: BuiltinCategoryKey }
export const DEFAULT_KEYWORD_RULES: readonly KeywordRule[]; // prototype list verbatim, then these additions:
// Subscriptions: netflix, spotify, disney, prime video, apple.com, icloud, youtube
// Groceries: tesco, sainsbury, asda, aldi, lidl, waitrose, morrisons, whole foods, safeway, kroger
// Transport: uber, lyft, train, rail, bus, parking, petrol, shell, bp
// Dining: restaurant, cafe, pizza, deliveroo, just eat, doordash, starbucks, pret
// Utilities: energy, gas, broadband, mobile, council
// Health: pharmacy, clinic, doctor, optician
// Income: salary, payroll, wages
export interface GuessContext {
  keywordRules: readonly KeywordRule[];
  learned: ReadonlyMap<string, string>;                    // normalised name -> category id
  builtinIds: ReadonlyMap<BuiltinCategoryKey, string>;      // active (non-archived) builtin -> id
}
export type GuessSource = 'learned' | 'keyword' | 'income-default' | 'none';
export function normaliseDescription(s: string): string; // NFD, strip \p{M}, lowercase, collapse whitespace, trim
export function buildLearnedMap(entries: readonly { name: string | null; categoryId: string | null; updatedAt: string }[]): Map<string, string>;
export function guessCategory(input: { name: string; amount: number }, ctx: GuessContext): { categoryId: string | null; source: GuessSource };
```
Matching: normalised name; learned exact match first; then keyword rules — collect all rules whose keyword occurs in the name, prefer whole-word matches (`\b` boundaries on the normalised string, keyword regex-escaped), then the longest keyword; map its category through builtinIds (skip if absent); then amount > 0 and builtinIds has Income -> income-default; else none.

```typescript
export interface DuplicateCandidate { index: number; localDate: string | null; amount: number | null; name: string }
export interface ExistingRow { id: string; localDate: string; amount: number; name: string | null }
export type DuplicateMatch = { kind: 'existing'; id: string } | { kind: 'in-file'; otherIndex: number };
export const NAME_SIMILARITY_THRESHOLD = 0.6;
export function nameSimilarity(a: string, b: string): number; // Jaccard over normalised word tokens; 1 when either normalised string contains the other
export function findDuplicates(candidates: readonly DuplicateCandidate[], existing: readonly ExistingRow[]): Map<number, DuplicateMatch>;
```
Rule: same localDate AND same amount (exact minor units) AND nameSimilarity >= 0.6 (existing null name counts as similarity 1 — Phase 1 rows had no name). Check existing first, then earlier candidates in the file (keyed by date|amount for O(n)). Candidates with null date/amount never match.

Create `src/engine/recurring/detect.ts`:
```typescript
export interface RecurringDetectRow { id: string; name: string; amount: number; currency: string; localDate: string }
export interface RecurringSuggestion { key: string; name: string; amount: number; currency: string; freq: RecurringFreq; anchorDate: string; rowIds: string[] }
export const AMOUNT_TOLERANCE = 0.1;
export const INTERVAL_WINDOWS: Readonly<Record<RecurringFreq, readonly [number, number]>> = {
  weekly: [6, 8], fortnightly: [12, 16], monthly: [27, 33], quarterly: [85, 97], yearly: [360, 370],
};
export const MIN_ROWS: Readonly<Record<RecurringFreq, number>> = { weekly: 3, fortnightly: 3, monthly: 3, quarterly: 2, yearly: 2 };
export function detectRecurring(rows: readonly RecurringDetectRow[]): RecurringSuggestion[];
```
Group by normaliseDescription(name) + currency + sign; sort each group by date; compute day gaps (UTC date arithmetic); pick the freq whose window contains every gap; require MIN_ROWS; median amount (lower median on even counts) and every amount within AMOUNT_TOLERANCE of it in absolute value; amount = median; anchorDate = latest date; key = `${normalised}|${currency}|${sign}`; name = the most recent row's raw name. Sort suggestions by rowIds.length desc then name.
Add exports to both barrels (`src/engine/csv/index.ts`: `export * from './duplicates'`; `src/engine/recurring/index.ts`: `export * from './detect'`). Commits test/feat.
