export interface TotalsInput { amountHome: number | null; status: TxStatus; isTransfer: boolean } // isTransfer = row.transfer_id !== null (D-50)
export interface MonthTotals { paidIn: number; paidOut: number; net: number; stillToCome: number; pendingCount: number; projectedCount: number; unconvertedCount: number; transferCount: number }
export function monthTotals(rows: readonly TotalsInput[], projections: readonly { amountHome: number | null }[]): MonthTotals;
// paid: positive -> paidIn, negative -> paidOut; pending + projections -> stillToCome (signed); skipped ignored;
// amountHome null on a paid/pending row or projection -> unconvertedCount++ and excluded from sums.
// isTransfer rows -> transferCount++ and excluded from every sum and count above, whatever their status (D-50).

// filters.ts
export interface ActivityFilter {
  categoryIds: readonly (string | null)[] | null; // null entry = uncategorised
  accountIds: readonly string[] | null;
  direction: 'all' | 'in' | 'out' | 'transfers'; // in/out exclude transfer legs; transfers = only transfer legs (D-50)
  amountMin: number | null;  // home-currency minor units, compared on magnitude
  amountMax: number | null;
}
export const EMPTY_FILTER: ActivityFilter = { categoryIds: null, accountIds: null, direction: 'all', amountMin: null, amountMax: null };
export interface FilterRow { category_id: string | null; account_id: string; original_amount: number; amountHome: number | null; name: string | null; note: string | null; transfer_id: string | null }

export function normaliseForSearch(s: string): string; // NFD, strip \p{M}, lowercase, collapse whitespace, trim
export function matchesSearch(row: Pick<FilterRow, 'name' | 'note'>, term: string): boolean; // every term token appears in name+note
export function filterRows<T extends FilterRow>(rows: readonly T[], f: ActivityFilter): T[];

// balance.ts
export interface BalanceLeg { currency: string; paidSum: string } // server returns sums as text (bigint-safe)
export interface AccountBalance { balance: number | null; otherCurrencies: { currency: string; paidSum: number }[]; overflow: boolean }
export function accountBalance(input: { openingBalance: number; currency: string; legs: readonly BalanceLeg[] }): AccountBalance;
// Sum with BigInt; if any result exceeds Number.MAX_SAFE_INTEGER in magnitude set overflow true and balance null.
