```typescript
export type UndoEntity = 'transactions' | 'categories' | 'recurring_series' | 'accounts';
export type PatchValue = string | number | boolean | null;
export const NOW_SENTINEL = '$now' as const; // server substitutes now() (plan 02-09)
export interface PatchOp { entity: UndoEntity; id: string; expectedVersion: number; patch: Readonly<Record<string, PatchValue>> }
export const UNDO_LABEL_KEYS = ['added','edited','deleted','deletedMany','markedPaid','markedPaidMany','markedUnpaidMany','skipped','imported','seriesCreated','seriesEdited','seriesEnded','categoryAdded','categoryEdited','categoryArchived','categoryMerged','accountAdded','accountEdited'] as const;
export type UndoLabelKey = (typeof UNDO_LABEL_KEYS)[number];
export interface UndoLabelParams { n?: number; name?: string }
export interface UndoStepDraft { id: string; labelKey: UndoLabelKey; labelParams: UndoLabelParams; ops: PatchOp[]; touchedIds: string[] }
export const MAX_UNDO_OPS = 6000;
export interface SeriesChangeSet {
  series: { id: string; versionAfter: number; before: Readonly<Record<string, PatchValue>> | null }; // null = the series was created
  inserted: readonly { id: string; version: number }[];        // occurrence rows the change created
  softDeleted: readonly { id: string; versionAfter: number }[]; // occurrence rows the change soft-deleted
  linked: readonly { id: string; versionAfter: number }[];      // existing rows the change linked to the series (D-09, D-21)
}
export type UndoStepStatus = 'available' | 'undone' | 'refused';
export interface UndoConflict { entity: UndoEntity; id: string; updatedBy: string | null; recordName: string | null; builtinKey: string | null; reason: 'changed' | 'not-found' }
```
