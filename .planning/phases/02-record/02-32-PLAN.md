```typescript
import { parseAmount, type ParseError } from './parseAmount';
import { type MinorUnits } from './types';
export type AmountMarker = 'none' | 'plus' | 'minus' | 'parens' | 'dr' | 'cr' | 'od';
export interface NumberNotation {
  decimal: '.' | ',';
  group: ',' | '.' | ' ' | "'";      // ' ' also accepts U+00A0/U+202F (parseAmount whitespace set); "'" also accepts U+2019
  grouping: 'western' | 'indian';     // chooses a file-neutral locale for parseAmount's group-placement check
}
export type NotatedAmountError = ParseError | 'conflicting-markers';
export type NotatedAmountResult =
  | { ok: true; magnitude: MinorUnits; marker: AmountMarker }
  | { ok: false; error: NotatedAmountError };
export function markerSign(m: AmountMarker): 1 | -1;
export function parseNotatedAmount(raw: string, notation: NumberNotation, exponent: number): NotatedAmountResult;
```
Algorithm (string scanning with index loops; the only regexes allowed are anchored, single-quantifier character classes such as `/\p{Sc}/u` on a single character — no nested quantifiers):
1. Strip leading/trailing whitespace including U+00A0, U+202F, U+2007 and zero-width U+200B/U+200C/U+200D/U+FEFF. Unwrap a spreadsheet `="…"` wrapper, then a plain `"…"` wrapper. Empty -> `empty`.
2. Collect markers from both ends repeatedly until nothing changes: a leading `-`, U+2212 or U+2013 -> minus; a trailing `-` -> minus; leading `+` -> plus; outer `(` … `)` -> parens; a leading or trailing word token `DR`/`Dr`/`dr`/`D` or `CR`/`Cr`/`cr`/`C` (bounded by whitespace, a digit or the string edge) -> dr/cr; trailing token `OD`/`od` -> od; a single currency symbol char (`/^\p{Sc}$/u` test on one char) at either end; a 3-letter upper-case ISO code token at either end separated by optional whitespace. Whitespace between tokens is skipped.
3. Resolve markers: minus-like = {minus, parens, dr, od}; plus-like = {plus, cr}. If both a minus-like and a plus-like marker were collected -> `conflicting-markers`. Otherwise the single strongest marker by precedence dr > cr > od > parens > minus > plus > none (so '-12.50 DR' -> dr).
4. Normalise U+2019 to `'`. Split the remaining digits at the LAST occurrence of `notation.decimal` only when that char is followed by digits only; if the fraction part is longer than `exponent`, strip trailing '0' characters down to `exponent`; if non-zero digits remain beyond `exponent` -> `too-many-decimals`.
5. Call `parseAmount(rest, { locale: notation.grouping === 'indian' ? 'en-IN' : 'en-US', exponent, separators: { decimal: notation.decimal, group: notation.group } })`. Map `{ ok: false, error }` straight through. The value is always >= 0 because every sign character was consumed in step 2 (any remaining sign char makes parseAmount return `invalid`).
6. Return `{ ok: true, magnitude, marker }`.
Never throw for input content; `parseAmount` throws only for an out-of-range exponent (programmer error, message has no input text).
