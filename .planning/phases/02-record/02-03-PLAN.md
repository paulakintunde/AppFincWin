export type ColumnRole = 'date' | 'description' | 'amount' | 'debit' | 'credit' | 'currency' | 'balance' | 'direction' | 'limit';
export interface ColumnMapping {
  date: number | null; description: number | null; amount: number | null;
  debit: number | null; credit: number | null; currency: number | null;
  balance: number | null; direction: number | null; limit: number | null;
}
export type MappingError = 'no-date' | 'no-description' | 'no-amount' | 'amount-and-debit-credit' | 'duplicate-column';
export const HEADER_KEYWORDS: Readonly<Record<ColumnRole, readonly string[]>>;
export function normaliseHeader(h: string): string; // trim, lower-case, NFD + strip \p{M}
export function balanceLabelOf(header: string): 'held' | 'owed' | 'available' | null;
export function detectColumns(header: readonly string[], sample: readonly (readonly string[])[]): { mapping: ColumnMapping; confidence: 'high' | 'low' };
export function validateMapping(m: ColumnMapping): MappingError[];
```
HEADER_KEYWORDS (matched against normaliseHeader(header); a header matches a role when it equals a keyword or contains it as a whole word):
- date: 'date', 'transaction date', 'posted', 'posting date', 'value date', 'booking date', 'datum', 'fecha', 'data', 'buchungstag'
- description: 'description', 'details', 'payee', 'merchant', 'narrative', 'memo', 'name', 'reference', 'transaction', 'omschrijving', 'beschreibung', 'concepto', 'libelle', 'verwendungszweck'
- amount: 'amount', 'value', 'transaction amount', 'bedrag', 'betrag', 'importe', 'montant', 'importo'
- debit: 'debit', 'paid out', 'money out', 'withdrawal', 'withdrawals', 'out', 'spent'
- credit: 'credit', 'paid in', 'money in', 'deposit', 'deposits', 'in', 'received'
- currency: 'currency', 'ccy', 'valuta', 'wahrung', 'moneda', 'devise'
- balance: 'balance', 'running balance', 'saldo', 'solde', 'kontostand', 'available', 'available credit', 'available balance', 'balance due', 'statement balance'
- direction: 'type', 'transaction type', 'dr/cr', 'debit/credit', 'direction'
- limit: 'credit limit', 'overdraft limit', 'limit'
Resolve roles in this order: limit, balance, direction, date, debit, credit, amount, currency, description — limit and balance first so 'Credit limit' and 'Available credit' are never taken as a credit column, direction before debit/credit so 'Debit/Credit' and 'DR/CR' headers become a direction column. Each column takes at most one role; the first matching column wins. If both debit and credit found, amount stays null. A direction column is only accepted when >= 80% of its non-empty samples are in the direction keyword table (see Task 2). balanceLabelOf: contains 'available' -> 'available'; contains 'owed' / 'balance due' / 'statement balance' / 'amount due' -> 'owed'; else null ('held' is never inferred from a header — the account kind decides, D-53).
Content sniffing for any role still null after header matching: date = first unused column where >= 80% of non-empty samples parse under some format of `parseCsvDate`; amount = first unused column where >= 80% of non-empty samples parse with `parseNotatedAmount` under either notationFor('.') or notationFor(','); description = first unused column with the longest average text length. confidence = 'high' when date, description and (amount or debit+credit) all came from headers, else 'low'.

import type { MinorUnits, NumberNotation } from '../money';
import type { StatementDraft, DraftRow } from '../statement';
export const MAX_NAME_LENGTH = 200; // = MAX_DESCRIPTION, mirrors the DB check added in 02-07
export const DIRECTION_OUT = ['debit', 'dr', 'sale', 'purchase', 'withdrawal', 'payment out', 'out'] as const;
export const DIRECTION_IN = ['credit', 'cr', 'payment', 'return', 'refund', 'deposit', 'in'] as const;
export interface CsvDraftOptions {
  mapping: ColumnMapping;
  dateFormat: DateFormat;
  notation: NumberNotation;
  delimiter: Delimiter;
  defaultCurrency: string;                        // target account currency (D-12)
  exponentFor: (code: string) => number | null;   // null = unknown currency
}
export function layoutSignatureOf(header: readonly string[], delimiter: Delimiter, decimal: '.' | ','): string;
export function csvToDraft(header: readonly string[], dataRows: readonly (readonly string[])[], opts: CsvDraftOptions): StatementDraft;
```
Per row: date via parseCsvDate (null -> 'bad-date'); description = cell trimmed, internal whitespace runs collapsed to one space, sliced to MAX_NAME_LENGTH ('' -> 'empty-description'); currency = mapped cell upper-cased if `exponentFor(code) !== null`, else defaultCurrency (+ 'unknown-currency' when the cell was non-empty); exponent = exponentFor(currency) (defaultCurrency is always known — throw RangeError('csvToDraft: default currency has no exponent') if not; that message carries no cell content). Amount: amount column -> parseNotatedAmount(cell, notation, exponent); debit/credit per the feature block (column-derived marker 'dr'/'cr'; an inner marker that contradicts the column -> 'conflicting-markers'); direction column maps its value through DIRECTION_OUT/DIRECTION_IN (normalised, whole-cell match) to 'dr'/'cr', combined with an unsigned amount; a contradicting signed amount -> 'conflicting-markers'. Parse error 'too-large' -> 'amount-too-large', 'conflicting-markers' -> 'conflicting-markers', others -> 'bad-amount'; magnitude 0 -> 'zero-amount'. rawAmount = the trimmed amount cell (or `debit` / `credit` cell used) sliced to MAX_RAW_CELL. Balance: parseNotatedAmount on the balance cell -> balanceMagnitude/balanceMarker; failure -> 'bad-balance'; rawBalance sliced to 64.
Summary lines (before indexing): a row whose normalised description starts with 'opening balance' or 'balance brought forward' (-> statedOpening) or 'closing balance' or 'balance carried forward' (-> statedClosing), OR a row with an empty amount (and empty debit/credit) but a readable balance (first such row -> opening if it precedes every transaction row, last such -> closing), is removed from rows; add warning 'summary-lines-removed'. Use `startsWith` on the normalised string (no regex).
File-level: source 'csv'; accountHint null; currency null; balanceLabel = balanceLabelOf(header[balance]) when a balance column is mapped; labels per the feature block; statedLimit = first readable limit cell magnitude or null; available null; periodStart/periodEnd = min/max row localDate; layoutSignature = layoutSignatureOf(header, delimiter, notation.decimal) = normalised header cells joined with '|' + '|' + delimiter + '|' + decimal, sliced to 500. Never throw or log for row content.
Export from the barrel and REMOVE any `parseSignedAmount`/`mapRows`/`negate` exports (they no longer exist). Commits `test(02-03): …` then `feat(02-03): map CSV rows into sign-free StatementDraft`.
