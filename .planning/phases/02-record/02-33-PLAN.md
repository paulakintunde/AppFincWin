export const OFX_LIMITS = { maxBytes: 5 * 1024 * 1024, maxElements: 20_000, maxDepth: 64 } as const;

export type OfxError = 'no-ofx-root' | 'too-large' | 'too-deep';

RED then GREEN. Create `src/engine/ofx/tree.ts`:
```typescript
export interface OfxNode { name: string; value: string | null; children: OfxNode[] }   // value set only on leaves
export type TreeWarning = 'malformed-close' | 'unclosed-aggregate';
export function buildOfxTree(tokens: readonly OfxToken[]): { ok: true; root: OfxNode; warnings: TreeWarning[] } | { ok: false; error: 'too-large' | 'too-deep' };
export function findAll(root: OfxNode, name: string): OfxNode[];          // iterative DFS, document order
export function childText(node: OfxNode, name: string): string | null;    // direct child leaf value
export function child(node: OfxNode, name: string): OfxNode | null;       // direct child element
```
Rule (ofxtools): an `open` immediately followed by a `text` token is a LEAF: create it with that value, attach to the current top of stack, and if the very next token is `close` of the same name, swallow it. An `open` followed by another `open`/`close` is an AGGREGATE: push it. On `close(X)`: if X is on the stack pop down to and including X (each aggregate popped above X on the way adds one 'unclosed-aggregate' warning); if X is not on the stack add 'malformed-close' and ignore. At EOF pop the rest with 'unclosed-aggregate' warnings (one per aggregate). Count elements; > OFX_LIMITS.maxElements -> too-large. Stack depth > OFX_LIMITS.maxDepth -> too-deep. The synthetic root is named '#root'. Warnings are a de-duplicated list of codes (no tag names, no text). No recursion anywhere (iterative loops only) so a deep hostile file cannot overflow the JS stack.
