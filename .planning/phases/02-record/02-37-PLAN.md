```typescript
export const TRANSFER_WINDOW_DAYS = 3;       // E5, D-52 "within a few days"
export const TRANSFER_TOLERANCE_BPS = 500;   // 5% for cross-currency legs (E5)
export const PAYMENT_KEYWORDS = ['PAYMENT', 'THANK YOU', 'TRANSFER TO', 'TRANSFER FROM', 'TFR', 'XFER', 'CARD PAYMENT', 'DIRECT DEBIT'] as const;
export interface ImportedLeg { index: number; accountId: string; localDate: string; amount: number; currency: string; name: string; trnType: string | null }
export interface ExistingLeg { id: string; accountId: string; localDate: string; amount: number; currency: string; name: string | null; paymentType: string | null; transferId: string | null }
export interface TransferAccount { id: string; name: string; kind: string; currency: string; exponent: number }
export type TransferSuggestion =
  | { kind: 'pair'; importIndex: number; existingId: string; score: number }
  | { kind: 'choose'; importIndex: number; options: string[] }
  | { kind: 'orphan-transfer'; importIndex: number };
export function isPaymentLike(name: string | null, trnType: string | null, paymentType: string | null): boolean;
export function daysBetween(a: string, b: string): number;   // |a - b| in days via Date.UTC on the parts (pure, no time zone)
export function matchTransfers(input: {
  imported: readonly ImportedLeg[]; existing: readonly ExistingLeg[]; accounts: readonly TransferAccount[];
  perEur: ReadonlyMap<string, ScaledRate>; windowDays?: number; toleranceBps?: number;
}): TransferSuggestion[];
```
Hard constraints, score and assignment exactly as RESEARCH §A5.2: candidates = existing legs with transferId null, accountId different from the imported leg's, opposite sign, daysBetween <= window, amount test (same currency: |a| === |b|; cross currency: convert |out| into the in-leg currency with convertMinor via both perEur rates and require `|conv − |in|| * 10000 <= tolBps * |in|` using BigInt; a missing rate or a RangeError skips the candidate). Score: Δdays 0 = 3, 1 = 2, 2–3 = 1; +2 if isPaymentLike on either side; +2 if the imported name (upper-cased) contains the other account's upper-cased name; +1 when the pair is deposit -> card with the card leg positive. Build all (importIndex, existingId) candidates, sort by (score desc, Δdays asc, amount difference asc, existingId asc, importIndex asc) — sorting candidates, not statement rows — then assign greedily one-to-one. Before assigning an imported row, if its best two remaining candidates tie on (score, Δdays, amount difference), emit 'choose' with both ids and assign neither. Imported rows left without a pair and with isPaymentLike true -> 'orphan-transfer'. Output sorted by importIndex. Barrel: append `export * from './match';`. Commit test then feat.
