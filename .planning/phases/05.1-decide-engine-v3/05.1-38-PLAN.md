```ts
export interface Due { readonly seriesId: string; readonly date: string }
export interface PaidLine { readonly id: string; readonly seriesId: string; readonly date: string }
/** Half the shortest gap between two occurrences, minus one day, capped at maxDays: windows of neighbouring occurrences never overlap. */
export function matchWindowDays(c: Cadence, maxDays: number): number // min(maxDays, floor((minPeriod - 1) / 2)), minPeriod = 7 * every (week) or 28 * every (month)
export function dueKey(d: Due): string // `${seriesId}|${date}`
export function settle(due: readonly Due[], lines: readonly PaidLine[], windowOf: (seriesId: string) => number): { readonly settled: ReadonlySet<string>; readonly usedLines: ReadonlySet<string> }
```

Deterministic and independent of input order. Write the tests first (RED), then implement. 100% coverage.
