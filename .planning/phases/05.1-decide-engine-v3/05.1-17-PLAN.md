```ts
export interface CashPoint { readonly date: string; readonly balance: MinorUnits }
export interface CashPathV3 { readonly asOf: string; readonly points: readonly CashPoint[] }
export interface DatedFlow { readonly date: string; readonly amount: MinorUnits }
export function spreadThrough(day: number, input: { readonly erest: MinorUnits; readonly e: MinorUnits; readonly asOf: string }): bigint // cumulative spread through day index `day` (0 = asOf)
export function buildCashPathV3(input: { readonly asOf: string; readonly days: number; readonly opening: MinorUnits; readonly series: readonly V3Series[]; readonly paidThisMonth: ReadonlySet<string> /* "seriesId|date" occurrences already paid */; readonly erest: MinorUnits; readonly e: MinorUnits; readonly extraOut: readonly DatedFlow[] }): CashPathV3
export function nextPayDate(series: readonly V3Series[], asOf: string): string | null // first confirmed live pay occurrence ≥ asOf
export type CashCheck = { readonly kind: 'ok'; readonly lowest: CashPoint } | { readonly kind: 'payday'; readonly until: string; readonly shortBy: MinorUnits } | { readonly kind: 'not_covered'; readonly shortBy: MinorUnits };
export function cashCheck(withPurchase: CashPathV3, payDate: string | null): CashCheck
```
