```ts
export interface IncomeInput { readonly asOf: string; readonly current: V3Month | null; readonly history: readonly V3Month[]; readonly series: readonly V3Series[]; readonly profile: IncomeProfile | null; readonly rough: RoughFigures | null; readonly custom: CustomOptions }
export function incomeThisMonth(input: IncomeInput): IncomeThisMonth

```
incomeThisMonth: received = sum of paid lines in the current month classified 'recurring_income' plus paid income lines whose resolved countsAs is 'available' (other income). dueConfirmed = for each live confirmed pay series (and available other-income series), occurrences in [month start, month end] with no paid line of that series on or after (occurrence − 3 days) — counted only when occurrence ≥ asOf − MISSED_PAY_GRACE_DAYS (else reason missed_pay_excluded). Profile rules applied on top as in behavior: steady replaces the total; match adds only `expected` pending lines; spread replaces lines that carry coversMonths by their per-month part (look back across `history` for spread lines whose cover window includes the current month); balance keeps received only. A mix: salary parts by the salary rule, other parts by `profile.variable` (null → as match). Pay groups: both series of a group are just two monthly series (24/yr, C2).
typicalIncome: steady → steady; else Σ over live confirmed pay series and available other-income series of `divideHalfUp(amount × num, 12 × den)`; if none and rough → rough.income with reason rough_figures_used; `hasConfirmedIncome` = any live confirmed pay series. All bigint, minorFromBigInt. Tests per behavior, every branch covered.
