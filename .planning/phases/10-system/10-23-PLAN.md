  alter table public.alert_outbox add column ticket_ids text[] not null default '{}', add column sent_at timestamptz;
  alerts_claim(p_limit int) returns setof jsonb  -- rows status 'queued' and deliver_after <= now(), oldest first, for update skip locked, set status 'sending', attempts+1; each jsonb carries outbox row + prefs (with defaults when missing) + tokens array for the user
  alerts_finish(p_id uuid, p_status text, p_delivery text, p_ticket_ids text[]) returns void  -- sets outbox status/sent_at/ticket_ids and alert_log.delivery/delivered_at
  alerts_hold(p_id uuid, p_until timestamptz) returns void  -- status back to 'queued', deliver_after = p_until, alert_log.held_for_quiet = true
  alerts_digest_users() returns setof jsonb  -- users with status 'digest' rows: user_id, prefs, tokens, counts by kind
  alerts_digest_done(p_user uuid, p_local_date date) returns void  -- digest rows -> 'sent', alert_log delivery 'digest', prefs.last_digest_date = p_local_date
  alerts_receipts_due() returns setof jsonb  -- sent rows with ticket_ids, sent_at between now()-24h and now()-15min, not yet checked (add column receipts_checked boolean default false)
  alerts_drop_token(p_token text) returns void
  cron: select cron.schedule('alerts-deliver-minute', '* * * * *', $$ select net.http_post(url := (select decrypted_secret from vault.decrypted_secrets where name = 'alerts_deliver_url'), headers := jsonb_build_object('Content-Type','application/json','x-alerts-secret',(select decrypted_secret from vault.decrypted_secrets where name = 'alerts_deliver_secret')), body := '{}'::jsonb, timeout_milliseconds := 30000); $$);

Expo message: { to, title: 'FincWin', body, channelId: 'alerts' | 'digest', ttl: 86400, data: { kind, logId } } — data never carries amounts or names.
