```ts
// pin.ts
export const PIN_LENGTH = 6;
export const LOCK_TIMING_MS = { now: 0, oneMinute: 60_000, fiveMinutes: 300_000, fifteenMinutes: 900_000 } as const;
export type LockTiming = keyof typeof LOCK_TIMING_MS;          // default 'oneMinute' (D-02)
export type Backoff = { kind: 'none' } | { kind: 'wait'; ms: number } | { kind: 'reauth' };
export function backoffFor(wrongTries: number): Backoff;        // 1-4 none, 5 30_000, 6 60_000, 7 120_000, 8 300_000, 9 900_000, >=10 reauth
export function isValidPin(pin: string): boolean;               // exactly 6 ASCII digits
export async function hashPin(pin: string, saltHex: string): Promise<string>;
export function constantTimeEqual(a: string, b: string): boolean;
// lockStore.ts (SecureStore keys prefixed 'fincwin.lock.'; settings JSON in AsyncStorage 'fincwin:lock-settings')
export interface LockSettings { enabled: boolean; timing: LockTiming; biometric: boolean; offerAnswered: boolean; ownerUserId: string | null }
export const DEFAULT_LOCK_SETTINGS: LockSettings; // { enabled: false, timing: 'oneMinute', biometric: false, offerAnswered: false, ownerUserId: null }
export interface LockState { settings: LockSettings; hasPin: boolean; wrongTries: number; lockUntil: number | null; backgroundedAt: number | null }
export async function loadLockState(): Promise<LockState>;
export async function saveLockSettings(patch: Partial<LockSettings>): Promise<LockSettings>;
export async function setPin(pin: string): Promise<void>;       // new salt, stores hash, clears tries and lockUntil
export async function checkPin(pin: string, now: number): Promise<{ ok: true } | { ok: false; backoff: Backoff; wrongTries: number }>;
export async function clearTries(): Promise<void>;
export async function markBackgrounded(now: number): Promise<void>;
export function shouldLockOnActive(s: LockState, now: number, coldStart: boolean): boolean; // enabled && (coldStart || backgroundedAt == null || now - backgroundedAt >= LOCK_TIMING_MS[timing])
export async function clearLock(): Promise<void>;               // deletes every lock key
export function registerLockWipe(): () => void;                 // registerWipeHandler({ id: 'appLock', wipe: clearLock })
// biometric.ts
export type BiometricKind = 'faceId' | 'touchId' | 'fingerprint' | 'none';
export async function biometricKind(): Promise<BiometricKind>;  // hasHardwareAsync && isEnrolledAsync, then supportedAuthenticationTypesAsync
export async function authenticateBiometric(promptMessage: string, cancelLabel: string): Promise<'ok' | 'cancel' | 'fail'>; // disableDeviceFallback: true, biometricsSecurityLevel: 'strong'
export function isBiometricPending(): boolean;                  // true while authenticateAsync runs (Pitfall 8)
// reauth.ts
export function decideAfterReauth(lockedUserId: string | null, signedInUserId: string): 'keep' | 'wipe'; // D-29: same -> keep, different or null -> wipe
```
