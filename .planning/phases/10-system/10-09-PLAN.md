```ts
export interface RpcClient { rpc(fn: string, args: Record<string, unknown>): Promise<{ data: unknown; error: unknown }> }
export type RateResult = { ok: true } | { ok: false; retryAfter: number };
export async function hitRateLimit(admin: RpcClient, key: string, limit: number, windowSeconds: number, now?: () => number): Promise<RateResult>;

export function tooManyRequests(retryAfter: number): Response; // 429, header Retry-After, body {"ok":false,"error":"rate_limited"}
```
