```ts
export interface AlertSpendLine { magnitude: MinorUnits; localDate: string /* YYYY-MM-DD */ } // expenses only, home currency, already converted
export interface SuggestInput {
  expenses: readonly AlertSpendLine[];
  today: string;                 // YYYY-MM-DD, person's local date
  homeCurrency: string;
  homeExponent: number;          // from resolveExponent (custom currencies pass their decimals)
  homePerEur: ScaledRate | null; // null when no rate is known
}
export interface SuggestedAmount { minor: MinorUnits; basis: 'history' | 'flat' }
export interface SuggestedAmounts { large: SuggestedAmount; low: SuggestedAmount }
export function suggestAlertAmounts(input: SuggestInput): SuggestedAmounts;
export const ALERT_HISTORY_MIN_EXPENSES = 30;
export const ALERT_HISTORY_MIN_MONTHS = 2;
export const FLAT_LARGE_EUR_MINOR = 20000; // 200.00 EUR
export const FLAT_LOW_EUR_MINOR = 10000;   // 100.00 EUR
```
Window: the 3 calendar months ending at `today` = local dates from (today minus 3 months, same day, exclusive) to today inclusive.
