  list_my_devices() returns table(session_id uuid, model text, os text, app_version text, last_active timestamptz, signed_in_at timestamptz, is_current boolean) — security definer; from auth.sessions s left join public.devices d on d.session_id = s.id where s.user_id = auth.uid(); last_active = greatest(d.last_active, s.refreshed_at, s.updated_at); signed_in_at = s.created_at; is_current = s.id = (auth.jwt()->>'session_id')::uuid.
