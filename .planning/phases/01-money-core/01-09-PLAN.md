export function subscribeFailedWrites(listener: () => void): () => void;

export function useSyncStatus(): SyncStatus {
  const isOnline = useSyncExternalStore(onlineManager.subscribe.bind(onlineManager), () => onlineManager.isOnline());
  const queued = useMutationState({ filters: { status: 'pending' } }).length; // paused mutations are status 'pending' with isPaused

  return { isOnline, queued, failed: failedList.filter((f) => f.kind !== 'conflict').length,

}
