1. Pure functions (`language sql immutable strict`, `parallel safe`):
```sql
create or replace function public.div_half_up(num numeric, den numeric) returns bigint
language plpgsql immutable strict set search_path = '' as $$
begin
  if den <= 0 then raise exception 'div_half_up: denominator must be positive' using errcode = '22012'; end if;
  return (sign(num) * div(2 * abs(num) + den, 2 * den))::bigint;
end $$;

create or replace function public.convert_minor(amount bigint, from_per_eur numeric, from_exp int, to_per_eur numeric, to_exp int) returns bigint
language sql immutable strict set search_path = '' as $$
  select public.div_half_up(
    amount::numeric * (to_per_eur * 10000000000) * power(10::numeric, to_exp),
    (from_per_eur * 10000000000) * power(10::numeric, from_exp))
$$;

create or replace function public.cross_rate(from_per_eur numeric, to_per_eur numeric) returns numeric(24,10)
language sql immutable strict set search_path = '' as $$
  select (public.div_half_up((to_per_eur * 10000000000) * 10000000000, from_per_eur * 10000000000)::numeric / 10000000000)::numeric(24,10)
$$;

create or replace function public.custom_per_eur(reference_per_eur numeric, unit_value numeric) returns numeric(24,10)
language sql immutable strict set search_path = '' as $$
  select (public.div_half_up((reference_per_eur * 10000000000) * 10000000000, unit_value * 10000000000)::numeric / 10000000000)::numeric(24,10)
$$;
```
`div()` on numeric truncates toward zero, and both operands are non-negative, so it is floor, matching BigInt division in the TS mirror. Note: `power(10::numeric, n)` returns numeric; never use `10 ^ n` (double precision).

3. `public.per_eur_rate(p_code text, p_on date, p_owner uuid, p_any_earlier boolean default false, out rate numeric, out rate_date date, out source text, out exact boolean)` (stable, security definer, plpgsql):
   - `p_code = 'EUR'` → rate 1, rate_date p_on, source null, exact true.
   - custom currency for p_owner → recurse for its reference_currency; if the reference result is null, return nulls; else rate = public.custom_per_eur(ref.rate, c.unit_value), rate_date = least(ref.rate_date, c.as_of), source = 'custom', exact = ref.exact.
   - ISO: `select r.rate, r.rate_date, r.source from public.fx_rates r where r.base = 'EUR' and r.quote = p_code and r.rate_date <= p_on and (p_any_earlier or r.rate_date >= p_on - 7) order by r.rate_date desc, (r.source = 'frankfurter-v2') desc limit 1` → exact true when found (D-02 nearest earlier; frankfurter preferred over open-er-api on the same date). If not found: nearest later `rate_date > p_on order by rate_date asc, (source='frankfurter-v2') desc limit 1` → exact false (provisional, D-17). Else, if p_any_earlier is false and an older-than-7-days earlier row exists, use it with exact false. Else nulls with exact false.
   - fx_rate_holds (plan 01-08) is a separate table and is never read here, so a held rate can never be served (D-11).

5. `public.stamp_fx_rate()` trigger function (plpgsql, security definer so it can read the author's profile and custom currencies even when another household member edits in Phase 8). Logic:
```
declare o record; h record; any_earlier boolean := coalesce(current_setting('fincwin.system_restamp', true), '') = 'on';
        owner uuid; needs_rerate boolean; o_exp int; h_exp int;
begin
  owner := coalesce(new.created_by, (select auth.uid()));
  if tg_op = 'INSERT' then
    select p.home_currency into new.home_currency from public.profiles p where p.id = owner;
    if new.home_currency is null then raise exception 'no profile for transaction author' using errcode = '23502'; end if;
  else
    new.home_currency := old.home_currency;             -- D-05: fixed at write time
  end if;
  needs_rerate := tg_op = 'INSERT' or new.local_date is distinct from old.local_date
                  or new.original_currency is distinct from old.original_currency
                  or old.rate_pending or any_earlier;
  if needs_rerate then
    if new.original_currency = new.home_currency then
      new.rate := 1; new.orig_per_eur := null; new.home_per_eur := null; new.rate_date := new.local_date;
      new.rate_source := 'same-currency'; new.rate_pending := false; new.home_amount := new.original_amount;
      return new;
    end if;
    select * into o from public.per_eur_rate(new.original_currency, new.local_date, owner, any_earlier);
    select * into h from public.per_eur_rate(new.home_currency, new.local_date, owner, any_earlier);
    if o.rate is null or h.rate is null then
      new.rate := null; new.orig_per_eur := null; new.home_per_eur := null; new.rate_date := null;
      new.rate_source := null; new.home_amount := null; new.rate_pending := true;
      return new;
    end if;
    o_exp := public.currency_exponent(new.original_currency, owner);
    h_exp := public.currency_exponent(new.home_currency, owner);
    new.orig_per_eur := o.rate; new.home_per_eur := h.rate;
    new.rate := public.cross_rate(o.rate, h.rate);
    new.rate_date := least(o.rate_date, h.rate_date);
    new.rate_source := case when 'open-er-api' in (o.source, h.source) then 'open-er-api'
                            when 'custom' in (o.source, h.source) then 'custom'
                            else 'frankfurter-v2' end;   -- attribution must never be lost (MON-12)
    new.rate_pending := not (o.exact and h.exact);
    new.home_amount := public.convert_minor(new.original_amount, o.rate, o_exp, h.rate, h_exp);
  else
    -- amount-only or note/account edit: keep every stamp column (D-04)
    new.rate := old.rate; new.orig_per_eur := old.orig_per_eur; new.home_per_eur := old.home_per_eur;
    new.rate_date := old.rate_date; new.rate_source := old.rate_source; new.rate_pending := old.rate_pending;
    if new.original_amount is distinct from old.original_amount then
      if old.rate_source = 'same-currency' then new.home_amount := new.original_amount;
      else new.home_amount := public.convert_minor(new.original_amount, old.orig_per_eur,
             public.currency_exponent(new.original_currency, owner), old.home_per_eur,
             public.currency_exponent(new.home_currency, owner));
      end if;
    else new.home_amount := old.home_amount;
    end if;
  end if;
  return new;
end;
```
(Handle the `'open-er-api' in (o.source, h.source)` null-safety: EUR contributes source null; write it as `o.source = 'open-er-api' or h.source = 'open-er-api'` so nulls are fine.)
`create trigger stamp_fx_rate before insert or update on public.transactions for each row execute function public.stamp_fx_rate();` Revoke execute on stamp_fx_rate from public, anon, authenticated.
