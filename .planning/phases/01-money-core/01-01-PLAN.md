```typescript
// src/engine/money/types.ts
export type MinorUnits = number & { readonly __brand: 'MinorUnits' };
export type CurrencyCode = string & { readonly __brand: 'CurrencyCode' };
export interface Money { readonly amount: MinorUnits; readonly currency: CurrencyCode }
export const MAX_ABS_AMOUNT_MINOR = 10_000_000_000_000; // 1e13, mirrors the DB check in 01-02
export function minorUnits(n: number): MinorUnits;          // throws RangeError unless Number.isSafeInteger(n)
export function currencyCode(s: string): CurrencyCode;      // /^[A-Z0-9]{2,4}$/ else RangeError (ISO 3-letter + custom 2-4 chars)
export function money(amount: number, currency: string): Money;

// src/engine/money/rounding.ts
export function divideHalfUp(numerator: bigint, denominator: bigint): bigint; // denominator must be > 0n; ties round away from zero, symmetric for negatives

// src/engine/money/rates.ts  -- a rate is "units of currency X per 1 EUR", scaled by 10^10 (fx_rates.rate is numeric(24,10))
export const RATE_SCALE = 10;
export type ScaledRate = bigint & { readonly __brand: 'ScaledRate' };
export const EUR_PER_EUR: ScaledRate;                       // parseRate('1')
export function parseRate(s: string): ScaledRate;           // /^\d{1,14}(\.\d{1,10})?$/, must be > 0, else RangeError
export function formatRate(r: ScaledRate): string;          // always 10 fraction digits: '1.1483000000' (matches Postgres numeric(24,10)::text)
export function convertMinor(amount: MinorUnits, fromPerEur: ScaledRate, fromExponent: number, toPerEur: ScaledRate, toExponent: number): MinorUnits;
//   = divideHalfUp(amount * toPerEur * 10^toExponent, fromPerEur * 10^fromExponent); throws RangeError if result is not a safe integer
export function crossRate(fromPerEur: ScaledRate, toPerEur: ScaledRate): ScaledRate;      // "1 from = x to" = divideHalfUp(toPerEur * 10^10, fromPerEur)
export function customPerEur(referencePerEur: ScaledRate, unitValue: ScaledRate): ScaledRate; // custom units per EUR = divideHalfUp(referencePerEur * 10^10, unitValue)

// src/engine/money/currencyExponents.ts
export const ISO_EXPONENT_EXCEPTIONS: Readonly<Record<string, number>>;
export function currencyExponent(code: string): number;                         // ISO table, default 2
export function resolveExponent(code: string, customDecimals?: number): number; // customDecimals (0..4) wins when provided

// src/engine/money/arithmetic.ts
export function add(a: Money, b: Money): Money;           // RangeError on currency mismatch or unsafe result
export function subtract(a: Money, b: Money): Money;
export function negate(m: Money): Money;
export function sum(items: readonly Money[], currency: CurrencyCode): Money; // every item must match currency
export function multiplyByInteger(m: Money, factor: number): Money;          // factor must be a safe integer
export function compare(a: Money, b: Money): -1 | 0 | 1;  // same-currency only
export function isZero(m: Money): boolean;
```
