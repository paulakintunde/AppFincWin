```typescript
// src/engine/payoff/types.ts
export type Percent = bigint & { readonly __brand: 'Percent' };
export const PERCENT_SCALE: bigint;            // 10n ** BigInt(RATE_SCALE)  === 10_000_000_000n
export const ONE_HUNDRED_PERCENT: bigint;      // 100n * PERCENT_SCALE
export const MONTHLY_RATE_DENOMINATOR: bigint; // 1200n * PERCENT_SCALE   (monthly rate r = apr / this)
export const ZERO_PERCENT: Percent;            // 0n
export function percent(s: string): Percent;   // '22.9' -> 229_000_000_000n; '0' -> 0n; '6' -> 60_000_000_000n
export function percentFromScaled(n: bigint): Percent; // RangeError when n < 0n; no upper bound
export function minorFromBigInt(value: bigint, fn: string): MinorUnits; // RangeError `${fn}: result exceeds Number.MAX_SAFE_INTEGER`
export function ceilDivide(n: bigint, d: bigint): bigint; // counts only (months needed); RangeError if d <= 0n or n < 0n
export interface AmortisationResult {
  readonly payment: MinorUnits;       // level monthly payment, months 1..n-1
  readonly finalPayment: MinorUnits;  // month n: remaining balance + that month's interest
  readonly months: number;
  readonly totalPaid: MinorUnits;     // === principal + totalInterest, exactly
  readonly totalInterest: MinorUnits;
}
export interface CardMinimumTerms {   // D-16: always from the card record, never an engine default
  readonly minimumPercent: Percent;
  readonly interestAdded: boolean;    // true: scheduled = percent-of-balance + this month's interest
  readonly floor: MinorUnits;         // in the card's own currency
}
export type MinimumOutcome = 'cleared' | 'never_clears' | 'exceeds_month_cap';
export interface MinimumPaymentResult {
  readonly outcome: MinimumOutcome;
  readonly months: number;            // cleared: months to clear; never_clears: 1-based month the check fired; exceeds_month_cap: SIMULATION_MONTH_CAP
  readonly firstPayment: MinorUnits;
  readonly totalInterest: MinorUnits | null; // null only for never_clears (unbounded)
  readonly remainingBalance: MinorUnits;     // 0 when cleared
}
// src/engine/payoff/constants.ts
export const SIMULATION_MONTH_CAP = 600; // D-17
export const MAX_TERM_MONTHS = 600;      // BUILD-PROMPT §4: dpmt tested up to n = 600
// src/engine/payoff/amortise.ts
export function amortise(principal: MinorUnits, apr: Percent, months: number): AmortisationResult;
// src/engine/payoff/futureValue.ts
export function futureValue(monthly: MinorUnits, apr: Percent, months: number, seed: MinorUnits): MinorUnits;
```
