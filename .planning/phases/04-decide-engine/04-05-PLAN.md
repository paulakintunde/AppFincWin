```typescript
import type { CurrencyCode, MinorUnits } from '../money/types';
import type { Percent, CardMinimumTerms, MinimumOutcome } from '../payoff/types';
import type { RecurringFreq } from '../recurring/schedule';
import type { ItemType, CardEligibility } from './cardRule';
import type { ShareScope, SplitRule, ResolvedShare } from './share';
export type { Percent, CardMinimumTerms, MinimumOutcome, ItemType, CardEligibility, ShareScope, SplitRule, ResolvedShare, RecurringFreq };

// ── Snapshot (in) — built by Phase 5 outside engine/ (DEC-12) ──
// D-16 clarified: one currency per check. The builder converts every figure into `DecideSnapshot.currency`
// before calling the engine; the engine rejects any other code (assertCurrency) and never converts.
export type LineStatus = 'pending' | 'paid' | 'skipped';   // Record row status (recurring/cloneMonth.ts CloneSourceRow.status)
export interface RecordLine {
  readonly id: string;
  readonly date: string;               // 'YYYY-MM-DD', inside its MonthRecord.month
  readonly direction: 'in' | 'out';
  readonly amount: MinorUnits;         // >= 0, in `currency`
  readonly currency: CurrencyCode;     // D-16: must equal DecideSnapshot.currency
  readonly status: LineStatus;         // D-28: only 'paid' lines count toward past months
  readonly isRefund: boolean;          // D-27: money in that nets against spending, never income; valid only with direction 'in'
  readonly transferId: string | null;  // a transfer leg is neither income nor spending
  readonly seriesId: string | null;    // Record recurring_series_id; lines of a smoothed series are replaced by its monthly equivalent (D-23)
  readonly categoryId: string | null;  // D-34 per-category share lookup; null → the household rule
}
// Income lines ('in', not a refund) are MY income. Spending lines ('out') are my own spending in personal scope and the
// household's shared spending in household scope; the engine applies my existing-spending share (D-34). The builder omits
// debt repayments and saving/investing contributions that are not transfer legs: those enter through debts[].minimumPayment,
// monthlySaving and monthlyInvesting.
export interface MonthRecord {
  readonly month: string;              // 'YYYY-MM'
  readonly full: boolean;              // D-13: set by the snapshot builder; the engine never infers it
  readonly lines: readonly RecordLine[];
}
export interface RecurringBill {       // a live recurring OUTFLOW series from Record; the builder omits ended series (their past lines then count as ordinary spending)
  readonly seriesId: string;
  readonly confirmed: boolean;         // D-25: created or accepted by the user; false (an unaccepted detect.ts suggestion) → ignored everywhere
  readonly freq: RecurringFreq;        // reused from src/engine/recurring/schedule.ts
  readonly anchorDate: string;         // schedule anchor, 'YYYY-MM-DD'
  readonly endDate: string | null;     // inclusive
  readonly occurrenceCount: number | null;
  readonly latestAmount: MinorUnits;   // D-26: the latest scheduled amount, not an average of past occurrences
  readonly currency: CurrencyCode;
  readonly categoryId: string | null;  // D-34
  readonly lastPaidDate: string | null; // D-24: date of the last occurrence with status 'paid'; null when none has been paid
}
export interface DebtRecord {
  readonly id: string; readonly balance: MinorUnits; readonly minimumPayment: MinorUnits;
  readonly secured: boolean;           // DEC-11
  readonly apr: Percent | null;        // D-31 only; null = not on record (never invented, D-07)
  readonly currency: CurrencyCode;
}
export interface CardRecord { readonly id: string; readonly apr: Percent | null; readonly minimumTerms: CardMinimumTerms | null; readonly currency: CurrencyCode; } // D-07 / D-16: null = not on record; floor already in the check's currency
export interface ExpenseLine { readonly lineId: string; readonly averageMonthly: MinorUnits; readonly cuttable: boolean; } // MY share (D-34) on the paid, refund-netted basis; cuttable=false for housing, saving, debt (decided by the builder from the category)
export type Basis = 'average' | 'worst';
export interface DecideLimits {
  readonly reserve: MinorUnits | null;             // D-01: null → DEFAULT_RESERVE_MONTHS × month's costs
  readonly dsrLine: Percent | null;                // D-04: null → DEFAULT_DSR_LINE (36%)
  readonly unsecuredIncomeMultiple: number | null; // D-04: null → DEFAULT_UNSECURED_INCOME_MULTIPLE (12)
  readonly investingFloor: MinorUnits | null;      // D-05: null → what the user invests now each month
  readonly savingFloor: MinorUnits | null;         // D-05: null → what the user saves now each month
  readonly spareFloor: MinorUnits | null;          // D-05: replaces the prototype's FLOOR = 25 for "too thin to be spare"; null → 0
}
export interface CategoryShareRule { readonly categoryId: string; readonly rule: SplitRule }
export interface HouseholdSpendingRule { readonly rule: SplitRule; readonly byCategory: readonly CategoryShareRule[] } // D-34: existing spending only
export interface DecideSnapshot {
  readonly currency: CurrencyCode;                 // D-16 clarified
  readonly asOf: string;                           // 'YYYY-MM-DD' check date; the engine has no clock (D-24 window)
  readonly months: readonly MonthRecord[];
  readonly bills: readonly RecurringBill[];        // D-23 / D-24 / D-25 / D-26
  readonly basisOverride: Basis | null;
  readonly cashOnHand: MinorUnits;                 // D-10: includes the current month's actual balances
  readonly debts: readonly DebtRecord[];
  readonly monthlySaving: MinorUnits;              // goal contributions; pausable (D-35)
  readonly monthlyInvesting: MinorUnits;           // automatic investing incl. recorded monthly investments (D-32); pausable (D-35)
  readonly scope: ShareScope;                      // PURCHASE share: rule = the per-check resolved split (D-15, D-34)
  readonly householdSpending: HouseholdSpendingRule | null; // EXISTING-spending share (D-34); required in household scope, ignored in personal scope
  readonly limits: DecideLimits;
  readonly projectionReturn: Percent | null;       // D-06: null → 0%
  readonly expenseLines: readonly ExpenseLine[];   // for the cut-a-line option (DEC-13)
}
// D-30: there is deliberately no field for past or open checks.

// ── Check (in) ──
export type CardPayoff = { readonly kind: 'full' } | { readonly kind: 'months'; readonly months: number } | { readonly kind: 'minimum' };
export type BuyPayment =
  | { readonly method: 'cash' }
  | { readonly method: 'loan'; readonly deposit: MinorUnits; readonly termMonths: number | null; readonly apr: Percent | null }
  | { readonly method: 'instalment'; readonly instalmentAmount: MinorUnits | null; readonly instalmentCount: number | null }
  | { readonly method: 'card'; readonly card: CardRecord | null; readonly payoff: CardPayoff; readonly dealerCap: MinorUnits | null }; // dealerCap: D-18, defaulted from the market table by the builder
export type InvestPayment = { readonly method: 'lump' } | { readonly method: 'monthly'; readonly termMonths: number | null };
export interface BuyCheck { readonly kind: 'buy'; readonly currency: CurrencyCode; readonly price: MinorUnits; readonly dayFees: MinorUnits; readonly itemType: ItemType; readonly runningCostsNet: MinorUnits; readonly payment: BuyPayment; }
export interface InvestCheck { readonly kind: 'invest'; readonly currency: CurrencyCode; readonly amount: MinorUnits; readonly payment: InvestPayment; }
export type PurchaseCheck = BuyCheck | InvestCheck;

// ── Plan (dplan) ──
export type PlanReason =
  | 'card_not_accepted' | 'card_missing' | 'card_apr_missing' | 'card_minimum_terms_missing' | 'dealer_cap_missing'
  | 'loan_term_missing' | 'loan_apr_missing' | 'instalment_amount_missing' | 'instalment_count_missing' | 'invest_term_missing';
export interface CardPlanDetail {
  readonly eligibility: Exclude<CardEligibility, 'blocked'>;
  readonly onCard: MinorUnits;          // prototype cardCap
  readonly remainderOnDay: MinorUnits;  // D-18: part above the dealer cap, paid as day-one cash
  readonly payoff: CardPayoff['kind'];
  readonly payoffMonths: number | null; // null when the minimum never clears
  readonly minimumOutcome: MinimumOutcome | null; // only for payoff 'minimum'
}
export interface Plan {
  readonly price: MinorUnits;
  readonly dayOne: MinorUnits;       // prototype `day`
  readonly monthly: MinorUnits;      // prototype `mo`
  readonly months: number;           // prototype `n`
  readonly financed: MinorUnits;     // prototype `fin` (adds to debt; 0 for invest)
  readonly total: MinorUnits | null; // null only when a card minimum never clears
  readonly interest: MinorUnits | null;
  readonly dayFees: MinorUnits;
  readonly card: CardPlanDetail | null;
}
export type PlanResult = { readonly status: 'resolved'; readonly plan: Plan } | { readonly status: 'missing'; readonly reasons: readonly PlanReason[] };

// ── Money basis (dmoney, plan 04-13) ──
export interface SpendingShare {     // D-34: resolved existing-spending share
  readonly byDefault: ResolvedShare;
  readonly byCategory: readonly { readonly categoryId: string; readonly share: ResolvedShare }[];
}
export type SpendingShareResult = { readonly ok: true; readonly share: SpendingShare } | { readonly ok: false; readonly reason: 'split_weights_zero' | 'split_amounts_zero' };
export interface MoneyInput {
  readonly currency: CurrencyCode;
  readonly months: readonly MonthRecord[];
  readonly bills: readonly RecurringBill[];
  readonly basisOverride: Basis | null;
  readonly spendingShare: SpendingShare;
}
export type Steadiness = 'steady' | 'variable' | 'volatile' | 'insufficient';
export interface MonthSummary { readonly month: string; readonly income: MinorUnits; readonly spending: MinorUnits } // my income; my net spending (refunds netted, floored at 0) plus smoothed bills
export interface MoneyBasis {
  readonly fullMonthsUsed: number;
  readonly monthsUsed: readonly MonthSummary[];   // newest first: the window actually averaged
  readonly incomeAverage: MinorUnits; readonly incomeWorst: MinorUnits; readonly incomeBest: MinorUnits;
  readonly spendingAverage: MinorUnits; readonly spendingWorst: MinorUnits; // my share; worst spending = highest month (D-08, D-12 amended)
  readonly smoothedBillsMonthly: MinorUnits;      // D-23: my share of the monthly equivalents, already inside every month's spending
  readonly steadiness: Steadiness;
  readonly basis: Basis;
  readonly incomeBasis: MinorUnits; readonly spendingBasis: MinorUnits;
}
export type MoneyResult = { readonly ok: true; readonly money: MoneyBasis } | { readonly ok: false; readonly reason: 'no_full_month' };

// ── Bills due (D-24) ──
export interface BillDue { readonly seriesId: string; readonly categoryId: string | null; readonly dates: readonly string[]; readonly amount: MinorUnits } // household amount = latestAmount × dates.length

// ── Verdict (dassess) ──
export type VerdictState = 'fits' | 'adjust' | 'no' | 'missing';
export type WarningCode = 'month' | 'cash' | 'dsr' | 'owe' | 'inv';
export interface Warning { readonly code: WarningCode; readonly shortfall: MinorUnits }
export type MissingReason = PlanReason | 'no_full_month' | 'split_weights_zero' | 'split_amounts_zero';
export type BasisMarker = 'logged_months' | 'first_read_rough_figures';
export type BasisFactCode = 'debt_rate_exceeds_return';
export interface BasisFact {         // D-31: neutral, non-blocking; never changes fits/adjust/no
  readonly code: 'debt_rate_exceeds_return';
  readonly debtId: string;           // the unsecured debt with the highest APR above the projection return
  readonly debtApr: Percent;
  readonly projectionReturn: Percent;
}
export interface ResolvedLimits {
  readonly reserve: MinorUnits; readonly dsrLine: Percent; readonly unsecuredLine: MinorUnits;
  readonly investingFloor: MinorUnits; readonly savingFloor: MinorUnits; readonly spareFloor: MinorUnits;
  readonly projectionReturn: Percent;
}
export interface LimitChecks { readonly cash: boolean; readonly month: boolean; readonly dsr: boolean; readonly inv: boolean; readonly save: boolean; }
export interface AssessContext {            // everything about the user that does not depend on the check
  readonly snapshot: DecideSnapshot;
  readonly money: MoneyBasis;
  readonly purchaseShare: ResolvedShare;     // D-15: the per-check resolved split; applied to the purchase only (D-34)
  readonly spendingShare: SpendingShare;     // D-34: my share of existing household spending and bills
  readonly limits: ResolvedLimits;
  readonly mySpendingBasis: MinorUnits;      // money.spendingBasis (already my share)
  readonly monthCosts: MinorUnits;           // D-02: money.spendingAverage (smoothed, D-23) + debt minimums
  readonly debtPayments: MinorUnits;
  readonly unsecuredDebt: MinorUnits;
  readonly pausable: MinorUnits;             // D-35: monthlySaving + monthlyInvesting (D-32)
  readonly leftBefore: MinorUnits;           // prototype M.left
  readonly billsDue: MinorUnits;             // D-24: my share of confirmed quarterly/yearly bills due within 90 days
  readonly billsDueDetail: readonly BillDue[];
  readonly cashAvailable: MinorUnits;        // cashOnHand − billsDue
  readonly highestUnsecuredApr: { readonly debtId: string; readonly apr: Percent } | null; // D-31 input
}
export type ContextResult = { readonly ok: true; readonly context: AssessContext } | { readonly ok: false; readonly reasons: readonly MissingReason[] };
export interface Assessment {
  readonly state: 'fits' | 'adjust' | 'no';
  readonly basisMarker: 'logged_months';
  readonly warnings: readonly Warning[];    // at most 5, shortfall desc, ties in WARNING_ORDER
  readonly facts: readonly BasisFact[];     // D-31; empty for buy checks
  readonly plan: Plan;
  readonly mineDayOne: MinorUnits; readonly mineMonthly: MinorUnits; readonly mineRunning: MinorUnits; readonly newMonthly: MinorUnits;
  readonly left: MinorUnits; readonly cashAfter: MinorUnits;   // cashAfter = cashAvailable − mineDayOne (D-24)
  readonly debtServiceAfter: MinorUnits; readonly unsecuredAfter: MinorUnits;
  readonly investingAtRisk: MinorUnits; readonly investingAfter: MinorUnits;  // D-35: paused from investing first
  readonly savingAtRisk: MinorUnits; readonly savingAfter: MinorUnits;        // D-35: then from saving
  readonly pauseNeeded: MinorUnits;         // investingAtRisk + savingAtRisk
  readonly remainingShortfall: MinorUnits;  // what the month still lacks after pausing (buy) or −left (invest, D-33)
  readonly burn: MinorUnits;                // monthly outgoings after this decision
  readonly checks: LimitChecks;
  readonly context: AssessContext;
}
export type Verdict = { readonly state: 'missing'; readonly reasons: readonly MissingReason[] } | Assessment;

// ── Bisection / options (DEC-09, DEC-13) ──
export type AffordablePrice = { readonly kind: 'full_price_fits'; readonly price: MinorUnits } | { readonly kind: 'nothing_affordable' } | { readonly kind: 'found'; readonly price: MinorUnits };
export type DecideOption =
  | { readonly code: 'affordable_price'; readonly price: MinorUnits; readonly reduction: MinorUnits }
  | { readonly code: 'longer_term'; readonly termMonths: number; readonly monthlyPayment: MinorUnits; readonly leftAfter: MinorUnits; readonly monthCloses: boolean; readonly leftChange: MinorUnits; readonly extraInterest: MinorUnits }
  | { readonly code: 'cut_line'; readonly lineId: string; readonly lineAverage: MinorUnits; readonly cut: MinorUnits; readonly gap: MinorUnits; readonly closesGap: boolean }
  | { readonly code: 'earn_more'; readonly gap: MinorUnits; readonly incomeBasis: MinorUnits; readonly basis: Basis; readonly percentOfIncome: Percent | null }
  | { readonly code: 'save_first'; readonly need: MinorUnits; readonly monthlySpare: MinorUnits; readonly months: number | null }
  | { readonly code: 'do_without'; readonly keptMonthly: MinorUnits; readonly horizonMonths: number; readonly projected: MinorUnits; readonly projectionReturn: Percent };
export type OptionCode = DecideOption['code'];

// ── Alternatives (DEC-14) ──
export type AlternativeCode = 'as_planned' | 'cheaper' | 'wait_and_save' | 'skip';
export type SuppressionCode = 'month_short' | 'too_thin' | 'nothing_freed' | 'past_horizon' | 'nothing_spare';
export interface Suppression { readonly code: SuppressionCode; readonly amount: MinorUnits | null }
export interface AlternativeRow {
  readonly code: AlternativeCode;
  readonly dayOne: MinorUnits | null; readonly monthly: MinorUnits | null;
  readonly contributed: MinorUnits | null; readonly projected: MinorUnits | null; readonly growth: MinorUnits | null; // null when suppressed or not applicable
  readonly suppression: Suppression | null;
}
export interface Alternatives {
  readonly horizonMonths: number; readonly projectionReturn: Percent; readonly cheaperPrice: MinorUnits;
  readonly spareCash: MinorUnits; readonly spareMonthly: MinorUnits;
  readonly monthsToBuy: number | null; readonly waitMonths: number;
  readonly rows: readonly [AlternativeRow, AlternativeRow, AlternativeRow, AlternativeRow]; // as_planned, cheaper, wait_and_save, skip
}

// ── Setbacks (DEC-15) ──
export type SetbackCode = 'unexpected_expense' | 'income_drop' | 'job_loss' | 'late_invoice';
export type SetbackInput =

  | { readonly code: 'income_drop'; readonly enabled: boolean; readonly percentDrop: number; readonly months: number }
```
