import type { CurrencyCode, MinorUnits } from '../money/types';
import type { Percent, CardMinimumTerms, MinimumOutcome } from '../payoff/types';
import type { RecurringFreq } from '../recurring/schedule';
import type { ItemType, CardEligibility } from './cardRule';
import type { ShareScope, SplitRule, ResolvedShare } from './share';
export type { Percent, CardMinimumTerms, MinimumOutcome, ItemType, CardEligibility, ShareScope, SplitRule, ResolvedShare, RecurringFreq };

// ── Snapshot (in) — built by Phase 5 outside engine/ (DEC-12) ──
// D-16 clarified: one currency per check. The builder converts every figure into `DecideSnapshot.currency`
// before calling the engine; the engine rejects any other code (assertCurrency) and never converts.
export type LineStatus = 'pending' | 'paid' | 'skipped';   // Record row status (recurring/cloneMonth.ts CloneSourceRow.status)
export interface RecordLine {
  readonly id: string;
  readonly date: string;               // 'YYYY-MM-DD', inside its MonthRecord.month
  readonly direction: 'in' | 'out';
  readonly amount: MinorUnits;         // >= 0, in `currency`
  readonly currency: CurrencyCode;     // D-16: must equal DecideSnapshot.currency
  readonly status: LineStatus;         // D-28: only 'paid' lines count toward past months
  readonly isRefund: boolean;          // D-27: money in that nets against spending, never income; valid only with direction 'in'
  readonly transferId: string | null;  // a transfer leg is neither income nor spending
  readonly seriesId: string | null;    // Record recurring_series_id; lines of a smoothed series are replaced by its monthly equivalent (D-23)
  readonly categoryId: string | null;  // D-34 per-category share lookup; null → the household rule
  readonly oneOff: boolean;            // D-47 (2026-10-15 expansion): user-marked "one-off, won't recur"; left out of the averages and the worst month and counted in MoneyBasis.oneOffsExcluded. Set by the builder from Record's row field (05 D-29)
}
// Income lines ('in', not a refund) are MY income. Spending lines ('out') are my own spending in personal scope and the
// household's shared spending in household scope; the engine applies my existing-spending share (D-34). The builder omits
// debt repayments and saving/investing contributions that are not transfer legs: those enter through debts[].minimumPayment,
// monthlySaving and monthlyInvesting.
export interface MonthRecord {
  readonly month: string;              // 'YYYY-MM'
  readonly full: boolean;              // D-13: set by the snapshot builder; the engine never infers it
  readonly lines: readonly RecordLine[];
}
export interface RecurringBill {       // a live recurring OUTFLOW series from Record; the builder omits ended series (their past lines then count as ordinary spending). Debt repayment series are never passed as bills: debts[].minimumPayment carries them, and the D-44 cash path would otherwise count them twice
  readonly seriesId: string;
  readonly confirmed: boolean;         // D-25: created or accepted by the user; false (an unaccepted detect.ts suggestion) → ignored everywhere
  readonly freq: RecurringFreq;        // reused from src/engine/recurring/schedule.ts
  readonly anchorDate: string;         // schedule anchor, 'YYYY-MM-DD'
  readonly endDate: string | null;     // inclusive
  readonly occurrenceCount: number | null;
  readonly latestAmount: MinorUnits;   // D-26: the latest scheduled amount, not an average of past occurrences
  readonly currency: CurrencyCode;
  readonly categoryId: string | null;  // D-34
  readonly lastPaidDate: string | null; // D-24: date of the last settled occurrence (status 'paid' or 'skipped'; a skipped occurrence settles that cycle, consistent with D-28 ignoring skipped lines); null when none has been settled. Set by the snapshot builder (Phase 5).
}
// D-44 (2026-10-15 expansion): a live confirmed PAY series (money in). Same schedule fields as a bill; income is mine and never shared,
// so there is no category. lastPaidDate = the last received (settled) occurrence. Only `confirmed: true` series count (D-25).
// D-60: a confirmed weekly or fortnightly pay series is smoothed in the income averages and the CV — in each month the series spans
// in full, its paid income lines (RecordLine.seriesId === seriesId) are replaced by monthlyEquivalent(latestAmount, freq) (D-26 latest
// amount); unconfirmed, monthly or other income is not smoothed. "Spans in full" (04-13, checker W2) reads the schedule fields above:
// anchorDate is the series' first scheduled pay date (occurrence n = 0 in schedule.ts), and the series ends at endDate or, with
// occurrenceCount, at its last scheduled date; a month the series starts late in or ends inside counts its logged lines as they are.
export type RecurringIncome = Omit<RecurringBill, 'categoryId'>;
// D-61 (2026-10-15): a dated pending one-off outflow, e.g. the pending expense Go ahead writes (05). Not a series, and not a Record line
// the averages read. `amount` is MY share, already resolved by the snapshot builder: the engine applies no split to it (unlike bills).
// It sits on its date in the D-44 cash path (dated on or before asOf → day 0; dated after asOf + 90 days → outside the path, ignored) and
// reduces the cash ledger (the `cash` check's cashAvailable and the D-38 Cash row's ledgerCash) only when dated on or before asOf, with or without a
// cash path; cashAfterBills (spare money) deducts every one dated within 90 days (series.ts scheduledOutflowsDue; AssessContext.outflowsDueNow / outflowsDueAhead).
export interface ScheduledOutflow {
